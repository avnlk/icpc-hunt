MOD = 998244353

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int,input().split()))
    
    h = [0]*(n+1)
    
    for i in range(1,n+1):
        h[i] = (h[i-1] + pow(i,MOD -2,MOD))%
MOD

    ans = 0
    
    for i in range(n):
      ans += a[i]*(h[i] + h[n-1-i])
      ans %= MOD
      
    print(ans)
