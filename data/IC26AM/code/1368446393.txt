#include <bits/stdc++.h>

#define int long long

const int mod=998244353;

int power(int x, int y){
    int result=1;
    int mul=x;
    while (y){
        if (y&1){
            result*=mul;
        }
        result%=mod;
        mul*=mul;
        mul%=mod;
        y>>=1;
    }
    return result;
}
int inv(int x){
    return power(x,mod-2);
}

void solve() {
    int n;
    cin >> n;
    vector<int> arr(n);
    int ans = 0;
    for (int &i:arr) cin >> i;
    vector<int> dp1(n+1), dp2(2*n+1);
    for (int i=1;i<=n;i++) {
        dp1[i] = dp1[i-1];
        dp1[i] += 2*inv(i);
        dp1[i] %= mod;
    }
    for (int i=1;i<=2*n;i++) {
        dp2[i] = dp2[i-1];
        dp2[i] += inv(i);
        dp2[i] %= mod;
    }
    for (int i=0;i<n;i++) {
        int left = i;
        int right = n-i-1;
        if (right < left) {
            swap(left, right);
        }
        right -= left;
        int curr = dp1[left];
        curr += dp2[left+right] - dp2[left];
        curr += mod;
        curr %= mod;
        // cout << curr << endl;
        curr *= arr[i];
        curr %= mod;
        ans += curr;
        ans %= mod;
    }
    cout << ans << '\n';
}
