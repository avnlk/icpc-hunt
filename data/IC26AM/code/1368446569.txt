#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;
int main() {
	// your code goes here  
	int t;
	cin>>t;
	vector<vector<long long>>tests(t);
	int mx = 1;
	for(int i=0;i<t;i++) {
	    int n;
	    cin>>n;
	    tests[i].resize(n);
	    for(int j=0;j<n;j++) {
	        long long x;
	        cin>>x;
	        x%=MOD;
	        if(x<0)x+=MOD;
	        tests[i][j] = x;
	    }
	    mx = max(mx,n);
	}
	vector<long long>inv(mx+2), H(mx+2);
	inv[1] = 1;
	for(int i=2;i<mx+1;i++) {
	    inv[i] = (MOD - (MOD/i) * inv[MOD%i]%MOD)%MOD;
	}
	H[0] = 0;
	for(int i=1;i<=mx+1;i++) {
	    H[i] = (H[i-1]+inv[i])%MOD;
	}
	for(int i = 0;i<t;i++) {
	    int n = tests[i].size();
	    long long ans = 0;
	    for(int j = 1;j<=n;j++) {
	        long long c = (H[j-1] + H[n-j])%MOD;
	        ans = (ans + tests[i][j-1]*c)%MOD;
	    }
	    cout<<ans<<'\n';
	}
}
