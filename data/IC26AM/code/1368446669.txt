#include <bits/stdc++.h>
using namespace std;

const long long m = 998244353;

int main() {
	// your code goes here
	const int n=200005;
	vector<long long> v(n),h(n);
	
	v[1]=1;
	for(int i=2;i<n;i++)
	{
	    v[i]=(m-(m/i) * v[m%i]%m)%m;
	}
    h[0]=0;
    for(int i=1;i<n;i++)
    {
        h[i]=(h[i-1]+v[i])%m;
    }
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        long long r=0;
        for(int i=1;i<=n;i++)
        {
            long long x;
            cin>>x;
            r=(r+x%m *((h[i-1] + h[n-i])%m))%m;
        }
        cout<<r<<endl;
    }
}
