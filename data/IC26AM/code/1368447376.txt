import sys
def main():
    MOD = 998244353
    data=sys.stdin.buffer.read().split()
    N=200005
    inv=[0,1]+[0]*(N-2)
    for i in range(2,N):
        inv[i]=(MOD-MOD//i)*inv[MOD%i]%MOD
    h=[0]*N
    for i in range(1,N):
        h[i]=(h[i-1]+inv[i])%MOD
    t=int(data[0])
    pos=1
    out=[]
    for _ in range(t):
        n=int(data[pos]);
        pos+=1
        ans=0
        for j in range(1,n+1):
            a=int(data[pos+j-1])
            ans+=a*(h[n-j]+h[j-1])
        pos+=n
        out.append(str(ans%MOD))
    print("\n".join(out))
main()
