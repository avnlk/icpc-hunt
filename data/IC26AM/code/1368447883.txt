import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc=new Scanner(System.in);
		int mod=998244353;
		
		int t=sc.nextInt();
		while(t-->0)
		{
		    int n=sc.nextInt();
		    long a[]=new long[n];
		    for(int i=0;i<n;i++)
		    {
		        a[i]=sc.nextLong();
		    }
		    
		    long inv[]=new long[n+1];
		    long h[]=new long[n+1];
		    
		    if(n>1)
		    {
		        inv[1]=1;
		    }
		    for(int i=2;i<=n;i++)
		    {
		        inv[i]=mod-(mod/i)*inv[mod%i]%mod;
		    }
		    for(int i=1;i<=n;i++)
		    {
		        h[i]=((h[i-1])+inv[i])%mod;
		    }
		    
		    long ans=0;
		    for(int i=0;i<n;i++)
		    {
		        long w=(h[i]+h[n-1-i])%mod;
		        
		        ans=(ans+a[i]*w)%mod;
		    }
		    
		    System.out.println(ans);
		    
		}

	}
}
