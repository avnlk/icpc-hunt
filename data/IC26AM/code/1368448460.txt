# cook your dish 
MOD = 998244353

t = int(input())
for _ in range(t):
    n = int(input())
    nums = list(map(int, input().split()))
    
    inv = [0] * (n - 1)
    inv = [0, 1] + inv
    
    for i in range(2, n + 1):
        inv[i] = (MOD - (MOD // i) * inv[MOD % i] % MOD) % MOD
        
    harm = [0] * (n + 1)
    for i in range(1, n + 1):
        harm[i] = (harm[i-1] + inv[i]) % MOD
    
    res = 0
    for i in range(1, n + 1):
        res += nums[i-1] % MOD * (harm[i-1] + harm[n-i])
    
    print(res % MOD)
