MOD = 998244353
MAX_N = 200000


def func():
    inv = [0] * (MAX_N + 1)
    H = [0] * (MAX_N + 1)
    
    inv[1] = 1
    
    for i in range(2, MAX_N + 1):
        inv[i] = MOD - (MOD//i) * inv[MOD%i] % MOD
    
    for i in range(1, MAX_N + 1):
        H[i] = (H[i-1] + inv[i] % MOD)
    
    return H 

def func2(a, H):
    n = len(a)
    ans = 0 
    
    for i in range(n):
        coe = H[i] + H[n - i - 1]
        coe %= MOD
        
        ans+=a[i] * coe
        ans%=MOD
    
    return ans

def func3():
    H = func()
    t = int(input())
    
    for _ in range(t):
        n = int(input())
        a = list(map(int, input().split()))
        print(func2(a, H))
        
func3()
