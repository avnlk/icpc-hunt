#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll MOD = 998244353;
const int MAXN = 200000 + 5;

ll inv[MAXN];
ll H[MAXN];

void precompute() {
    inv[1] = 1;
    
    for(int i = 2; i < MAXN; i++) {
        inv[i] = MOD - (MOD / i) * inv[MOD % i] % MOD;
    }
    
    H[0] = 0;
    
    for(int i = 1; i < MAXN; i++) {
        H[i] = (H[i-1] + inv[i] % MOD);
    }
}

void solve() {
    int n;
    cin >> n;
    
    ll ans = 0;
    
    for(int i = 1; i <= n; i++) {
        ll x;
        cin >> x;
        
        ll coeff = (H[i-1] + H[n-i]) % MOD;
        
        ans = (ans + x % MOD * coeff) % MOD;
    }
    
    cout << ans << endl;
}

int main() {
    precompute();
    
    int t;
    cin >> t;
    
    while(t--) {
        solve();
    }
    
    return 0;

}
