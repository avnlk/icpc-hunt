#include <bits/stdc++.h>
using namespace std;
#define int long long

const int MOD= 998244353;
int power(int b,int e){
    int r=1;
    b%=MOD;
    while(e>0){
        if(e&1)r=r*b%MOD;
        b=(b*b)%MOD;
        e>>=1;
    }
    return r;
}


signed main() {
	// your code goes here
	int t; cin>>t;
	while(t--){
	    int n; cin>>n;
	    vector<int>arr(n,0);
	    for(int i=1;i<=n;i++){
	        arr[i]= (arr[i-1]+power(i,MOD-2))%MOD;
	    }
	    int ans=0;
	    for(int i=1;i<=n;i++){
	        int a; cin>>a;
	        ans=(ans+a*((arr[i-1]+arr[n-1])%MOD))%MOD;
	    }
	    cout<<ans<<endl;
	}
	return 0;

}
