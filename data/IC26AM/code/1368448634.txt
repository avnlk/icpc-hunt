#include <bits/stdc++.h>
using namespace std;


	// your code goes here
	const long long MOD = 998244353;
	long long in[200005];
	long long sum[200005];
	
	int main(){
	 
	    int t ;
	    cin>>t;
	    in[1] = 1;
	    sum[0] =0;
	    for(int i =1;i<=200000;i++){
	        if(i>1){
	            in[i]=MOD-(MOD/i)*in[MOD%i]%MOD;
	            
	        }
	        sum[i]=(sum[i-1]+in[i])%MOD;
	    }
	    while(t--){
	       int n;
	       cin>>n;
	       long long ans =0;
	       
	       for(int i = 0;i<n;i++){
	           long long x ;
	           cin>>x;
	           long long count = (sum[i]+sum[n-i-1])%MOD;
	           ans = (ans+x*count)%MOD;
	           
	       }
	       
	       cout<<ans<<"\n";
	       
	       
	    }
	    return 0;
	}
