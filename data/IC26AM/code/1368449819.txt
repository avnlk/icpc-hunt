import sys 
input = sys.stdin.readline
MOD = 998244353
MAXN = 200000
inv = [0] * (MAXN + 1)
harm = [0] * (MAXN + 1)
inv[1] = 1
for i in range(2, MAXN + 1):
    inv[i] = MOD - (MOD // i) * inv[MOD % i] % MOD
for i in range(1, MAXN + 1):
    harm[i] = (harm[i - 1] + inv[i]) % MOD 
t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    answer = 0
    for i in range(n):
        coef = (harm[i] + harm[n - i - 1]) % MOD
        answer = (answer + a[i] * coef) % MOD
    print(answer)
