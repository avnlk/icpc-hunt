#include <bits/stdc++.h>
using namespace std;

using ll = long long;
using vll = vector<long long>;
static const ll MOD = 998244353;
const int MX = 200005;

int solve(vll &inv, vll &H)
{
    ll n, ans = 0;
    cin>>n;
    vll a(n);
    for(ll i=1;i<=n;i++)
    {
        ll val;
        cin>>val;
        ll coefficient = (H[i-1]+H[n-i]) % MOD;
        ans = (ans + val % MOD * coefficient) % MOD;
    }
    
    cout<<ans<<endl;
    return 0;
}

int main() {
	// your code goes here
   int t;
   cin>>t;
    vll inv(MX+1) , H(MX+1,0);
    inv[1] = 1;
    for(ll i=2;i<=MX;i++)
       inv[i] = (MOD-(MOD/i)*inv[MOD%i]%MOD)%MOD;
    
    for(ll i=1;i<=MX;i++)
       H[i] = (H[i-1]+inv[i])%MOD;
       
   while(t--)
   {
       solve(inv,H);
   }
}
