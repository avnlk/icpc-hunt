#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll mod = 998244353;
const int n = 200005;

int main() {
	// your code goes here
  vector <ll> inv(n),h(n);
  inv[1]=1;
  for(int i = 2; i<n;i++){
      inv[i] = (mod - mod/i)*inv[mod % i] % mod;
      h[0]=0;
      
  }
  for(int i = 1; i< n;i++){
      h[i] = (h[i-1]+inv[i])% mod;
      
  }
  int t ;
  cin>>t;
  while(t--){
      int n;
      cin>>n;
      ll ans =0;
      for(int j =1; j<=n;j++){
          ll a;
          cin>>a;
          ans=(ans+a%mod*((h[j-1] +h[n-j])%mod))%mod;
      }
      cout<<ans<<endl;
  }
}
