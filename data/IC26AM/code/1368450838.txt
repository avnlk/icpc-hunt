#include <bits/stdc++.h>
using namespace std;

int main() {
    long long t, m = 998244353;
    auto p = [&](auto& f, long long b, long long e) -> long long {
        return e ? (e & 1 ? b * f(f,b, e - 1) % m : f(f, b * b % m, e / 2)) : 1;
    };
    auto s = [&](){
        int n;
        cin >> n;
        vector <long long> a(n), h(n, 0);
        for(long long &x : a) cin >> x;
        for(int i = 1; i < n; i++) h[i] = (h[i - 1] + p(p, i, m - 2)) % m;
        long long z = 0;
        for(int i = 0; i < n; i++) z = (z + a[i] * (h[i] + h[n - 1 - i]) % m) % m;
        cout << z << endl;
    };
    for(cin >> t; t--;) s();
}
