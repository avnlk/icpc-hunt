#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;
const long long N = 200005;
int main() {
    
    
	// your code goes here
	std::vector<long long> a(N), b(N);
    a[1] = 1;
    for (int i=2; i<N; i++){
        a[i] = (MOD - (MOD / i) * a[MOD % i]%MOD) %MOD;
    }
    b[0]=0;
    for(int i=1; i<N; i++){
        b[i] = (b[i-1]+ a[i]) %MOD;
    }
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        
        long long ans = 0;
        
        for (int i=1; i<=n; i++){
            long long temp;
            cin >> temp;
            long long c = (b[i-1]+ b[n-i]) % MOD;
            ans = (ans + temp % MOD * c) % MOD;
        }
        cout << ans << "\n";
    }
}
