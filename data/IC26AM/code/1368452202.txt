#include <bits/stdc++.h>
using namespace std;
#define ll long long
ll M=998244353;
ll pow(ll a, ll b){
    ll r=1;
    while(b){
        if(b&1==1) r=r*a%M;
        a=a*a%M;
        b>>=1;
    }
    return r;
}
ll inv(ll x){
    return pow(x,M-2);
}
int main() {
	// your code goes here
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        vector<ll>a(n);
        for(int i=0; i<n; i++) cin>>a[i];
        vector<ll>h(n+1,0);
        for(int i=1; i<n; i++){
            h[i]=h[i-1]+inv(i);
            h[i]%=M;
        }
        int ans=0;
        for(int i=0; i<n; i++){
            int temp=h[i]+h[n-i-1];
            temp%=M;
            ans+=a[i]*temp%M;
            ans%=M;
        }
        cout<<ans<<endl;
    }
}
