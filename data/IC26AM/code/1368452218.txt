vi hn(200010, 0);

int mod = 998244353;
int powMod(int x, int n){
    int ans = 1;
    int p = x;
    while(n > 0){
        if(n & 1){
            ans = (ans * p) % mod;
        }
        p = p * p % mod;
        n /= 2;
    }
    return ans;
}

int modinv(int x){
    return powMod(x, mod-2);
}

void solve(){
    int n; cin >> n; 
    vi a(n+1);
    forr(i,1,n) cin >> a[i];
    int ans = 0;
    forr(i,1,n){
        ans += (a[i] * ((hn[i-1] + hn[n-i])%mod))%mod;
        ans %= mod;
    }cout << ans << '\n';
}
