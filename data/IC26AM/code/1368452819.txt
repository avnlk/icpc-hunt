import sys 

def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
     
    t = int(input_data[0])
    out=[]
     
    MOD = 998244353
    MAXN = 200005
     
    inv = [0] * MAXN
    if MAXN > 1:
         inv[1] = 1
    for i in range(2, MAXN):
        inv[i] = (MOD - MOD // i) * inv[MOD % i] % MOD
    
    H = [0] * MAXN
    for i in range(1, MAXN):
        H[i] = (H[i-1] + inv[i]) % MOD
    idx = 1 
    for n in range(t):
        n = int(input_data[idx])
        idx += 1 
        a = []
        for _ in range(n): 
            a.append(int(input_data[idx]))
            idx += 1 
        ans = 0 
        for j in range(n): 
            coeff = (H[j] + H[n-1-j]) % MOD 
            ans = (ans + a[j] * coeff) % MOD
        out.append(str(ans))
    print('\n'.join(out))
    
if __name__ == "__main__":
    solve()
