#include <bits/stdc++.h>

using namespace std;
using ll = long long;

ll MOD = 998244353;
ll MAXN = 200000;

int main() {
    vector < ll > inv(MAXN + 1);
    vector < ll > harmonic(MAXN + 1);
    inv[1] = 1;
    for (int i = 2; i <= MAXN; i++) {
        inv[i] = MOD - (MOD / i) * inv[MOD % i] % MOD;

    }
    for (int i = 1; i <= MAXN; i++) {
        harmonic[i] = (harmonic[i - 1] + inv[i]) % MOD;
    }
    ll t;
    cin >> t;
    while (t--) {
        
        ll n;
        cin >> n;
        ll ans = 0;
        for (int i = 0; i < n; i++) {
            ll x;
            cin >> x;
            ll times = (harmonic[i] + harmonic[n - i - 1]) % MOD;

            ans = (ans + x * times) % MOD;
        }
        cout << ans << "\n";
    }
}
