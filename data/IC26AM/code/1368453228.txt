import sys
MOD = 998244353
def solve():
    data=list(map(int,sys.stdin.buffer.read().split()))
    ptr=0
    t=data[ptr]
    ptr+=1
    tests=[]
    max_n=0
    for _ in range(t):
        n=data[ptr]
        ptr+=1
        a=data[ptr:ptr+n]
        ptr+=n
        tests.append((n,a))
        max_n=max(max_n,n)
        inv=[0]*(max_n+1)
        if max_n>=1:
            inv[1]=1
            for i in range(2,max_n+1):
                inv[i]=MOD-(MOD//i)*inv[MOD%i]%MOD
                H=[0]*(max_n+1)
                for i in range(1,max_n+1):
                    H[i]=(H[i-1]+inv[i])%MOD
            for n,a in tests:
                ans=0
                for i in range(n):
                    coeff=(H[i]+H[n-1-i])%MOD
                    ans=(ans+a[i]*coeff)%MOD
                print(ans)
    solve()
