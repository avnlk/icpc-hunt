#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;
const int N = 200005;
long long inv[N], H[N];
int main() {
	// your code goes here
	inv[1] = 1;
	for(int i = 2; i < N ; i++) inv[i] = (MOD - MOD / i * inv[MOD % i] % MOD ) % MOD;
	for(int i = 1; i < N; i++) H[i] = (H[i - 1] + inv[i]) % MOD;
	int t;
	cin >> t;
	while(t--){
	    int n;
	    cin >> n;
	    long long ans = 0;
	    for(int j = 1; j <= n; j++){
	        long long a;
	        cin >> a;
	        ans = (ans + a * ((H[j - 1]) + H[n - j]) % MOD) % MOD;
	    }
	    cout << ans << endl;
	}

}
