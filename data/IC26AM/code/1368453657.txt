#include <bits/stdc++.h>
using namespace std;
#define int long long

const int mod = 998244353;
int modinv(int x){
    x%=mod;
    int need = mod-2;
    int ans = 1;
    for(int i = 0;i<30;i++){
        if((1<<i)&need){
            ans*=x;
            ans%=mod;
        }
        x*=x;
        x%=mod;
    }
    return ans;
}

void solve(){
    int n;
    cin >> n;
    int arr[n];
    for(int i = 0;i<n;i++){
        cin >> arr[i];
    }
    int fac[n+5];
    int inv[n+5];
    fac[0]=inv[0]=1;
    for(int i = 1;i<n+5;i++){
        fac[i]=i*fac[i-1];
        fac[i]%=mod;
        inv[i]=modinv(fac[i]);
    }
    int sum = 0;
    int pref[n];
    pref[0]=arr[0];
    for(int i = 1;i<n;i++){
        pref[i]=pref[i-1]+arr[i];
        pref[i]%=mod;
    }
    int ans = 0;
    for(int len = 1;len<=n-2;len++){
        int x = len-1;
        sum+=pref[n-1-len]-pref[len-1];
        sum%=mod;
        int curr = sum*2*fac[x];
        curr%=mod;
        curr*=inv[x+2];
        curr%=mod;
        ans+=curr;
        ans%=mod;
    }
    //prefix and suffix now
    for(int len = 1;len<n;len++){
        int x = len-1;
        int sum = pref[x]+pref[n-1]-pref[n-1-len];
        sum%=mod;
        int curr = sum*fac[x];
        curr%=mod;
        curr*=inv[x+1];
        curr%=mod;
        ans+=curr;
        ans%=mod;
    }
    ans+=mod;
    ans%=mod;
    cout << ans << "\n";
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    int t;
    cin >> t;
    while(t--){
        solve();
    }
    return 0;
}
