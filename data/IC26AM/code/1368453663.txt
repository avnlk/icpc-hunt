#include <bits/stdc++.h>
using namespace std;

typedef long long ll;
const ll MOD= 998244353;


int main() {
	// your code goes here
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	const int N=200005;
	vector<ll>inv(N+1),H(N+1,0);
	inv[1]=1;
	for(int i=2; i<=N; i++)
	  inv[i]=(MOD-(MOD/i)*inv[MOD%i]%MOD)%MOD;
	for(int i=1; i<=N; i++)
	  H[i]=(H[i-1]+inv[i])%MOD;
	  
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    ll ans =0;
	    for(int j=1; j<=n; j++){
	        ll a;
	        cin>>a;
	        ll c=(H[j-1]+H[n-j])%MOD;
	        ans = (ans+a%MOD*c)%MOD;
	    }
	    cout<<ans<<"\n";
	}
	return 0;
}
