#include <bits/stdc++.h>
using namespace std;

#define ll long long
const ll MOD = 998244353;

ll binexp(ll n, ll m){
    n %= MOD;
    ll ans = 1;
    while(m>0){
        if(m%2==1){
            ans = (ans*n)%MOD;
        }
        n = (n*n)%MOD;
        m>>=1;
    }
    return ans;
}
ll inv(ll x){
    return binexp(x,MOD-2);
}


    
void solve() {
    ll n;
    cin >> n;
    
    vector<ll> a(n);
    for (ll i=0; i<n; i++) cin >> a[i];
    
    vector<ll> fact(n+1);
    fact[0] = 1;
    for (ll i=1; i<=n; i++) {
        fact[i] = (fact[i-1] * i) % MOD;
    }

    vector<ll> inv_fact(n+1);
    inv_fact[n] = inv(fact[n]);
    for(ll i=n-1; i>=1; i--){
        inv_fact[i] = (inv_fact[i+1]*(i+1))%MOD;
    }
    inv_fact[0]=1;
    vector<ll> pre(n+1);
    pre[0] = 1;
    for (ll i=1; i<=n; i++) {
        pre[i] += pre[i-1] + inv_fact[i];
        pre[i] %= MOD;
    }
    
    vector<ll> dp(n+1);
    dp[0] = 0;
    ll sum = 0;
    for (ll i=1; i<=n; i++) {
        dp[i] = (sum*inv(i))%MOD;
        dp[i]++;
        dp[i] %= MOD;
        
        sum += dp[i];
        sum %= MOD;
    }
    
    
    ll ans = 0;
    
    for (ll i=0; i<n; i++) {
        ll temp = dp[i] + dp[n-1-i];
        temp %= MOD;
        temp *= a[i];
        temp %= MOD;
        
        ans += temp;
        ans %= MOD;
    }
    
    cout << ans << endl;
    
}
int main() {
	// your code goes here
	
	int t;
	cin >> t;
	
	while (t--) solve();

}
