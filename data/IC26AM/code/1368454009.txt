#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll MOD = 998244353;

int main() {
int t;
cin>>t;
while(t--){
    int n;
    cin>>n;
    vector<ll>a(n+1),inv(n+1),h(n+1);
    
    for(int i =1;i<=n;i++){
        cin>>a[i];
    }
    inv[1]=1;
    for(int i =2;i<=n;i++)
    inv[i]= MOD-(MOD/i)*inv[MOD%i]%MOD;
    
    for(int i=1;i<=n;i++){
        h[i]=(h[i-1]+inv[i])%MOD;
    }
    ll ans=0;
    for(int i=1;i<=n;i++){
        ll coef=(h[i-1]+h[n-i])%MOD;
        ans=(ans+a[i]%MOD*coef)%MOD;
    }
    cout<<ans<<endl;
}

}
