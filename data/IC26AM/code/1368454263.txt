#include <bits/stdc++.h>
using namespace std;

const long long mod = 998244353 ;

vector<long long> findh(int n){
    vector<long long> inverse(n+1);
    vector<long long> h(n+1);
    
    inverse[1] = 1 ;
    for(int i=2 ; i<=n ; i++){
        inverse[i] = (mod - mod / i) * inverse[mod%i]%mod;
    }
    for(int i=1 ; i<=n ; i++){
        h[i] = (h[i-1] + inverse[i])%mod ;
    }
    return h ;
}

int main() {
	// your code goes here
	int t ;
	cin >> t ;
	
	while(t--){
	    int n ;
	    cin >> n ;
	    
	    vector<long long> arr(n);
	    for(int i=0 ; i<n ; i++){
	        cin >>arr[i];
	    }
	    
	    vector<long long> h = findh(n);
	    long long answer = 0 ;
	    for(int i=0 ; i<n ; i++){
	        long long coef = (h[i] + h[n-1-i]) %mod ;
	        answer = (answer + coef * arr[i]) % mod ;
	    }
	    cout << answer << endl ;
	}

}
