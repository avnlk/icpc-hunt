import sys
MOD = 998244353
input = sys.stdin.buffer.readline
t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    inv = [0] * (n + 1)
    if n:
        inv [1] = 1
        for i in range(2, n + 1):
            inv[i] = MOD - (MOD // i) * inv[MOD % i] % MOD
    h = [0] * (n + 1)
    for i in range(1, n + 1):
        h[i] = (h[i - 1] + inv[i]) % MOD
    ans = 0
    for i, x in enumerate(a):
        ans = (ans + x * (h[i] + h[n - i -1])) % MOD
    print(ans)
