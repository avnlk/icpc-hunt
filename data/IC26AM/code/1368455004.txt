#include <bits/stdc++.h>
using namespace std;
const long long MOD=998244353;
const int MAXN=200000;

int main() {
	// your code goes here
	vector<long long> inv(MAXN +1), pref(MAXN+1);
	inv[1]=1;
	for(int i=2;i<=MAXN;++i){
	    inv[i]= MOD-(MOD/i)*inv[MOD%i]%MOD;
	}
	for(int i=1;i<=MAXN;++i)
	    pref[i]=(pref[i-1]+inv[i])%MOD;
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    long long ans=0;
	    for(int i=1;i<=n;++i){
	        long long x;
	        cin>>x;
	        x%=MOD;
	        ans=(ans+x*((pref[i-1]+pref[n-i])% MOD))%MOD;
	    }
	    cout<<ans<<endl;
	}
	return 0;

}
