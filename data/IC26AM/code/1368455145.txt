# cook your dish here
import sys
MOD=998244353
x =list(map(int,sys.stdin.buffer.read().split()))
t =x[0]
p=1 
ns =[]
aa =[] 
mx =0 
for _ in range(t):
    n =x[p] 
    p+=1 
    a =x[p:p+n]
    p+=n  
    ns.append(n)
    aa.append(a)
    mx =max(mx,n)
iv =[0]*(mx+1)
h =[0]*(mx+1)
if mx>=1:
    iv[1]=1


for i in range(2, mx+1):
    iv[i]=MOD-(MOD//i)* iv[MOD%i]%MOD
for i in range(1,mx+1):
     h[i]=(h[i-1]+iv[i])% MOD
ans =[]
for n,a in zip(ns,aa):
    s=0
    for i in range(n):
        c=(h[i]+h[n-1-i])%MOD
        s=(s+a[i]*c)%MOD
    ans.append(str(s))
sys.stdout.write("\n".join(ans))
