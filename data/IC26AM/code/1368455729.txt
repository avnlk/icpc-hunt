import sys
input = sys.stdin.readline

MOD = 998244353
n= 200000

h = [0]*(n+1)
for i in range(1,n+1):
    h[i]= (h[i-1]+pow(i,MOD-2,MOD))

t = int(input())
out =[]
for _ in range(t):
    n = int(input())
    a = list(map(int,input().split()))
    ans =0
    for i in range(n):
        ans = (ans+a[i]*(h[i]+h[n-1-i]))%MOD
    out.append(ans)
    
print("\n".join(map(str,out)))
