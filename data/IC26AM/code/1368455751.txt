# cook your dish here
MOD=998244353
MAXN=200005
inv=[0]*MAXN
inv[1]=1
for i in range(2, MAXN):
    inv[1]=(MOD-MOD//i)*inv[MOD%i]%MOD
H=[0]*MAXN
for i in range(1,MAXN):
    H[i]=(H[i-1]+inv[i])%MOD
t=int(input())

for _ in range(t):
    n=int(input())
    if n==1:
        print("0")
    a=list(map(int,input().split()))
    ans=0 
    for j in range(n):
        coeff=(H[j]+H[n-1-j])
        ans=(ans+a[j]*coeff)%MOD 
    print(str(ans))
