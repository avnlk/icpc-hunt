# cook your dish here
mod=998244353
m=200000

inv=[0,1]+[0]*(m-1)

for i in range(2,m+1):
    inv[i]=mod-(mod//i)*inv[mod%i]%mod
    
h=[0]*(m+1)

for i in range(1,m+1):
    h[i]=(h[i-1]+inv[i])%mod
    
t=int(input())

for _ in range(t):
    n=int(input())
    a=list(map(int,input().split()))
    
    ans=0 
    
    for i in range(n):
        ans+=a[i]*(h[i]+h[n-1-i])
        
    print(ans%mod)
