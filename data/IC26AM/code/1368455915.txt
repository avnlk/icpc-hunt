# cook your dish here
mod=998244353
t=int(input())
for _ in range(t):
    n=int(input());
    a=list(map(int,input().split()))
    h=[0]*n
    for i in range(1,n):
        h[i]=(h[i-1]+pow(i,mod-2,mod))%mod
    ans=0
    for i in range(n):
        cow=(h[i]+h[n-1-i])%mod
        ans=(ans+a[i]*cow)%mod
    print(ans)
