t = int(input())
while t:
    t-=1
    n = int(input())
    a = list(map(int,input().split()))
    
    MOD = 998244353
    def inv(x):
        p = MOD-2
        res = 1
        while p:
            if p&1: res = res*x%MOD 
            x = x*x%MOD 
            p>>=1
        return res
       
    ans = 0
    pre = a[0]
    for i in range(1,n-1):
        pre+=a[i]
        ans+=pre*inv(i+1)%MOD
        ans%=MOD
    suf = a[-1]
    for i in range(2,n):
        suf+=a[n-i]
        ans+=suf*inv(i)%MOD 
        ans%=MOD
    if n>1: ans=(ans+sum(a))%MOD
    
    pre = a.copy()
    for i in range(1,n):
        pre[i]+=pre[i-1]
    ppre = pre.copy()
    for i in range(1,n):
        ppre[i]+=ppre[i-1]
    
    for k in range(2,n-1):
        ans+=(ppre[n-2]-ppre[k-1]-ppre[n-k-2])*2*(inv(k)*inv(k+1)%MOD)%MOD
        ans%=MOD
    print(ans)
