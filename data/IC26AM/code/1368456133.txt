#include <bits/stdc++.h>
using namespace std;
using int64 = long long;
const int64 MOD = 998244353;

int main() {
    int T;
    cin>>T;
    // while(T){
        
    //     T--;
    // }
    
    vector<vector<int64>> tests(T);
    vector<int> ns(T);
    
    int maxN = 0;
    
    for(int tc = 0; tc < T; tc++){
        int n;
        cin>>n;
        
        ns[tc] = n;
        maxN = max(maxN,n);
        
        tests[tc].resize(n);
        for(auto &x: tests[tc]){
            cin>>x;
        }
    }
    
    vector<int64> inv(maxN+1,0);
    
    if(maxN>=1) inv[1]=1;
    
    for(int i=2;i<=maxN;i++){
        inv[i] = MOD -(MOD/i) * inv[MOD %i] % MOD;
    }
    
    vector<int64> H(maxN +1,0);
    
    for(int i=1;i<=maxN;i++){
        H[i] = (H[i-1] + inv[i]) %MOD;
        
    }
    
    for(int tc=0;tc<T; tc++){
        int n=ns[tc];
        const auto &a =tests[tc];
        
        int64 ans=0;
        
        for(int i=1;i<=n;i++){
            int64 exp_dep=(H[i-1] + H[n-i]) % MOD;
            ans=(ans + (a[i-1] % MOD)* exp_dep)%MOD;
        }
        cout<<ans<<endl;
    }
    return 0;
}
