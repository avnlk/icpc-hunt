#include <bits/stdc++.h>
using namespace std;

const long long MOD=998244353;

int main() {
	// your code goes here
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >>t;
    while (t--){
        int n;
        cin >>n;
        vector<long long> a(n+1);
        for (int i=1; i<=n; i++){
            cin>> a[i];
            a[i] %=MOD;
            if (a[i]<0) a[i]+=MOD;
        }
        
        vector<long long>inv(n+2, 1), H(n+1,0);
        for (int i=2; i<=n; i++)
            inv[i]=(MOD - (MOD/i)* inv[MOD%i]%MOD)%MOD;
        for(int i=1; i<=n; i++)
            H[i]=(H[i-1]+inv[i])%MOD;
            
        long long ans=0;
        for (int j=1; j<=n; j++){
            long long coef= (H[j-1]+ H[n-j])%MOD;
            ans=(ans+a[j]*coef)%MOD;
        }
        cout <<ans<<"\n";
    }
    return 0;
}
