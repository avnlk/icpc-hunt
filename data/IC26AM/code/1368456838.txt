#include <stdio.h>
#define MOD 998244353
#define MAX 200005

long long inv[MAX],h[MAX];

void precompute()
{
    inv[1]=1;
    for(int i = 2;i<MAX;i++)
    inv[i] = MOD-(MOD/i)*inv[MOD%i]%MOD;
    
    h[0] = 0;
    for(int i=1;i<MAX;i++)
    h[i] = (h[i-1]+inv[i])%MOD;
}

int main() {
	int t,n;
	scanf("%d",&t);
	
	precompute();
	
	while(t--)
	{
	    scanf("%d",&n);
	    long long ans = 0;
	    for(int i=1;i<=n;i++){
	        long long x;
	        scanf("%lld",&x);
	        long long weight = (h[i-1]+h[n-i])%MOD;
	        ans = (ans+(x%MOD)*weight)%MOD;
	    }
	    printf("%lld\n",ans);
	}
	return 0;
	

}
