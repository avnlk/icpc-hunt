#include <bits/stdc++.h>
using namespace std;
const long long MOD=998244353;
const int MAXN=200005;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	vector<long long>inv(MAXN),h(MAXN);
	inv[1]=1;
	for(int i=2;i<MAXN;i++){
	    inv[i]=MOD-(MOD/i)*inv[MOD%i]%MOD;
	}
	for(int i=1;i<MAXN;i++){
	    h[i]=(h[i-1]+inv[i])%MOD;
	}
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    vector<long long>a(n);
	    for(int i=0;i<n;i++){
	        cin>>a[i];
	    }
	    long long ans=0;
	    for(int i=0;i<n;i++){
	        long long c=(h[i]+h[n-1-i])%MOD;
	        ans=(ans+a[i]*c)%MOD;
	    }
	    cout<<ans<<'\n';
	}
	return 0;
}
