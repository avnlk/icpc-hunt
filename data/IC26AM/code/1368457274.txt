#include <bits/stdc++.h>
using namespace std;
const long long MOD=998244353;
int main() {
	// your code goes here
	const int N=200005;
	vector<long long> inv(N+1),H(N+1,0);
	inv[1]=1;
	for(int i=2;i<=N;i++)
	{
	    inv[i]=(MOD-(MOD/i)*inv[MOD%i]%MOD)%MOD;
	}
	for(int i=1;i<=N;i++)
	{
	    H[i]=(H[i-1]+inv[i])%MOD;
	}
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin>>n;
        long long ans=0;
        for(int j=1;j<=n;j++)
        {
            long long a;
            cin>>a;
            long long w=(H[j-1]+H[n-j])%MOD;
            ans=(ans+a%MOD*w)%MOD;
        }
        cout<<ans<<endl;
    }
    return 0;
}
