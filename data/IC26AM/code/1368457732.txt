typedef long long ll;
const ll MOD = 998244353;
int main() {
	// your code goes here
    const int N = 200005;
    vector<ll> inv(N + 1), H(N + 1);
    inv[1] = 1;
    for(int i = 2;i <= N ; i++) {inv[i] = (MOD - (MOD / i) * inv[MOD % i] % MOD) % MOD;
    H[i] = (H[i - 1] + inv[i] % MOD);}
    for(int i = 1; i <=N;i++) H[i] = H[i -1] + inv[i] % MOD;
    int t;
    scanf("%d",&t);
    while(t--){
        int n;
        scanf("%d",&n);
        ll ans = 0;
        for(int j = 1;j <= n ;j++){
            ll a;
            scanf("%lld", &a);
            ll w = (H[j -1] + H[n - j]) % MOD;
            ans = ( ans + a % MOD * w) % MOD;
        }
        printf("%lld\n", ans);
    }
    return 0;
}
