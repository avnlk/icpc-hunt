#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	const int MOD=998244353;
	const int MAXN=200005;
	static long long inv[MAXN];
	static long long H[MAXN];
	inv[1]=1;
	for(int i=2;i<MAXN;i++){
	    inv[i]=MOD-(MOD/i)*inv[MOD%i]%MOD;
	    
	}
	H[0]=0;
	for(int i=1;i<MAXN;i++){
	    H[i]=(H[i-1]+inv[i])%MOD;
	}
	int t;
	if(cin>>t){
	    while(t--){
	        int n;
	        cin>>n;
	        vector<long long>a(n);
	        long long ans=0;
	        for(int i=0;i<n;i++){
	            cin>>a[i];
	            long long exp_cnt=(H[i]+H[n-1-i])%MOD;
	            ans=(ans+a[i]*exp_cnt)%MOD;
	        }
	        cout<<ans<<endl;
	    }
	    
	}
	return 0;

}
