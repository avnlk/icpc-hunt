#include <bits/stdc++.h>
using namespace std;

const long long MOD=998244353;


int main() {
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    
	    vector<long long>a(n);
	    
	    for(int i=0;i<n;i++){
	        cin>>a[i];
	    }
	    
	    vector<long long>v(n+1,0);
	    vector<long long>v2(n+1,0);
	    
	    if(n>=1){
	        v[1]=1;
	        
	    }
	    for(int i=2;i<=n;i++){
	        v[i]= MOD-(MOD/i)* v[MOD%i]%MOD;
	    }
	    for(int i=1;i<=n;i++){
	        v2[i]=(v2[i-1]+ v[i])%MOD;
	    }
	    
	    long long op=0;
	    for(int i=0;i<n;i++){
	        long long l=v2[i];
	        long long r=v2[n-1-i];
	        long long cf=(l+r)%MOD;
	        op=(op+(a[i]%MOD)*cf)%MOD;
	    }
	    cout<<op<<endl;
	}
    return 0;
}
