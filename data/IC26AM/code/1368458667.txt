#include <bits/stdc++.h>
using namespace std;


using ll=long long;
const ll MOD=998244353;

ll modpow(ll a, ll b){
    ll res=1;
    
    while(b>0){
        if(b&1){
            res=res*a%MOD;
        }
        
        a=a*a%MOD;
        b>>=1;
    }
    return res;
}

int main() {
	// your code goes here
	int t;
	cin>>t;
	
	const int MAXN=200000;
	
	vector<ll> inv(MAXN+1);
	vector<ll> h(MAXN+1,0);
	
	inv[1]=1;
	
	for(int i=2;i<=MAXN;i++){
	    inv[i]=MOD-(MOD/i)*inv[MOD%i]%MOD;
	}
	
	for(int i=1;i<=MAXN;i++){
	    h[i]=(h[i-1]+inv[i])%MOD;
	}
	
	while(t--){
	    int n;
	    cin>>n;
	    
	    ll ans=0;
	    
	    for(int i=0;i<n;i++){
	        ll x;
	        cin>>x;
	        
	        ll coefficient= (h[i]+h[n-1-i])%MOD;
	        
	        ans=(ans+(x%MOD)*coefficient)%MOD;
	    }
	    
	    cout<<ans<<endl;
	}
	
	return 0;
}
