MOD=998244353
t=int(input())
for _ in range(t):
    n=int(input())
    a=list(map(int,input().split()))
    H=[0]*n 
    for i in range(1,n):
        H[i]=(H[i-1] + pow(i,MOD-2,MOD))%MOD
    ans=0
    
    for i in range(n):
        weight=(H[i]+H[n-1-i])%MOD
        ans=(ans+a[i]*weight)%MOD
    print(ans)
