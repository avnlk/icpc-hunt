#include <bits/stdc++.h>
using namespace std;

using ll = long long;
const ll MOD = 998244353;
const int MAXN = 200000;

int main() {
    int t;
    cin>>t;
	vector<int>n(t);
	vector<vector<ll>>a(t);
	int mx=0;
	for(int tc=0;tc<t;tc++)
	{
	    cin>>n[tc];
	    mx = max(mx,n[tc]);
	    a[tc].resize(n[tc]);
	    
	    for(int i = 0;i<n[tc];i++){
	        cin>>a[tc][i];
	    }
	}
	vector<ll>inv(mx+1,0);
	if(mx>=1){
	    inv[1] = 1;
	    for(int i = 2;i<=mx;i++){
	        inv[i] = MOD-(MOD/i)*inv[MOD%i]%MOD;
	    }
	}
	vector<ll>H(mx+1,0);
	for(int i = 1;i<=mx;i++){
	    H[i] = (H[i-1] +inv[i])%MOD;
	}
	for(int tc = 0;tc<t;tc++){
	    int N = n[tc];
	    ll ans = 0;
	    for(int i = 0;i<N;i++){
	        ll coef = (H[i] + H[N-1-i])%MOD;
	        ans+=a[tc][i]*coef;
	        ans%=MOD;
	    }
	    cout<<ans<<"\n";
	}
	
return 0;
}
