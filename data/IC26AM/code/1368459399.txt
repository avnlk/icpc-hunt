#include <bits/stdc++.h>
using namespace std;
using int64 = long long;
const int64 MOD = 998244353;
const int MAXN = 200000;
int main() {
	// your code goes here
    vector<int64> inv(MAXN+1),h(MAXN+1);
    inv[1] = 1;
    
    for(int i=2;i<=MAXN;++i)
        inv[i] =MOD-(MOD/i)*inv[MOD%i]%MOD;
    for(int i =1;i<=MAXN;++i)
        h[i] = (h[i-1] + inv[i]) % MOD;
        
        int t;
        cin>>t;
        
        while(t--){
            int n;
            cin>>n;
            int64 ans = 0;
            for(int i=0; i<n; i++){
                int64 x;
                cin>>x;
                int64 w = (h[i] + h[n-1-i]) % MOD;
                ans = (ans+x*w)%MOD;
            }
            cout<<ans<<'\n';
        }
    return 0;
}
