# cook your dish here
MOD = 998244353
mn = 2 * 10 ** 5 + 5
inv = [0] * mn
if mn > 1:
    inv[1] = 1
for i in range(2, mn):
    inv[i] = (MOD - MOD // i) * inv[MOD % i] % MOD
H = [0] * mn
for i in range(1, mn):
    H[i] = (H[i - 1] + inv[i]) % MOD
t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    res = 0
    for k in range(n):
        ck = (H[k] + H[n - 1- k]) % MOD
        res = (res + a[k] * ck) % MOD
    print(res)
