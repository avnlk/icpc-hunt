import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
    static final long MOD = 998244353;
    static long pow (long a , long b){
        long res = 1;
        while(b>0){
            if((b & 1) == 1){
                res = res * a % MOD;
                
            }
            a = a*a % MOD;
            b >>= 1; 
        }
        return res;
    }
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc = new Scanner(System.in);
		int s = sc.nextInt();
		while(s-- > 0 ){
		    int n = sc.nextInt();
		    long[] a = new long[n];
		    for(int i=0;i<n;i++){
		        a[i] = sc.nextLong();
		    }
		    long[] h = new long[n+1];
		    h[0] = 0;
		    for(int i=1;i<=n;i++){
		        h[i] = (h[i-1] + pow(i,MOD-2)) % MOD;
		}
		long ans =0;
		for(int i=0;i<n;i++){
		    long q = (h[i] + h[n-i-1]) % MOD;
		    ans = (ans + a[i] * q) % MOD;
		}
		System.out.println(ans);

	
		}
		}
}
