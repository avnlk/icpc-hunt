import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		while(t-- > 0){
		    int n = sc.nextInt();
		    List<Integer> a = new ArrayList<>();
		    for(int i = 0; i < n; i++){
		        int in = sc.nextInt();
		        a.add(in);
		    }
		    long s = 0;
		    int i = 0;
		    while(a.size()-1>1){
		        int temp = a.get(i) + a.get(i+1);
		        a.remove(i);
		        a.remove(i+1);
		        a.set(i, temp);
		        s += temp;
		    }
		    
		    long m = s%998244353;
		    System.out.println(m);
		}

	}
}
