import sys
def main():
    mod=998244353
    data=sys.stdin.buffer.read().split()
    t=int(data[0])
    N=200001
    inv=[0,1]+[0]*(N-1)
    for i in range(2,N+1):
        inv[i]=(mod-(mod//i)*inv[mod%i]%mod)%mod
    h=[0]*(N+1)
    for i in range(1,N+1):
        h[i]=(h[i-1]+inv[i])%mod
    pos=1
    out=[]
    for _ in range(t):
        n=int(data[pos]);pos+=1 
        tot=0 
        for i in range(n):
            tot+=int(data[pos+i])*(h[i]+h[n-1-i])
        pos+=n 
        out.append(tot%mod)
    print("\n".join(map(str,out)))
main()
