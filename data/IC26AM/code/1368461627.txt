#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353;
const int MAXN = 200005;
ll inv [MAXN], H[MAXN];
int main() {
	// your code goes here
	ios_base::sync_with_stdio(false);
	cin.tie(0);
	inv[1]=1;
	for(int i=2;i<MAXN;i++){
	    inv[i]=(MOD-(MOD/i)*inv[MOD%i]%MOD)%MOD;
	}
	H[0]=0;
	for(int i =1;i<MAXN;i++){
	    H[i]=(H[i-1]+inv[i])%MOD;
	}
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    vector<ll>a(n+1);
	    for(int i=1;i<=n;i++){
	        cin>>a[i];
	    }
	    ll answer = 0;
	    for(int k=1;k<=n;k++){
	        ll depth=(H[k-1]+H[n-k])%MOD;
	        answer=(answer+a[k]%MOD*depth)%MOD;
	    }
	    cout<<answer<<endl;
	}
	return 0;
}
