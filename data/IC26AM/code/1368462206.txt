#include <stdio.h>

#define MOD 998244353LL
#define MAXN 200005
long long inv [MAXN], H[MAXN];


int main() {
	// your code goes here
inv[1]=1;
for(int i = 2; i<MAXN; i++)
inv[i]= (MOD - (MOD/i)*inv[MOD %i]%MOD) %MOD;
H[0] =0;
for(int i= 1; i<MAXN; i++)
H[i]= (H[i-1]+inv[i])%MOD;
int t;
scanf("%d", &t);
while(t--){
    int n;
    scanf("%d",&n);
    long long ans = 0;
    for(int j= 1; j<=n;j++){
        long long a;
        scanf("%lld",&a);
        long long c = (H[j-1]+H[n-j]) %MOD;
        ans = (ans + a %MOD*c)%MOD;
        
    }
    
    printf("%lld\n",ans);
}
return 0;
}
