#include <bits/stdc++.h>
#define ll long long
#define MOD 998244353
using namespace std;

ll modPow(ll a,ll b,ll mod){
    ll res = 1;
    while(b > 0){
        if(b&1) res = (res*a)%mod;
        a = (a*a)%mod;
        b >>= 1;
    }
    return res;
}
ll modInverse(ll a, ll mod){
    return modPow(a,mod-2,mod);
}
const int M = 2e5+1;
vector<int> har(M);
void pre(){
    har[1] = 1;
    for (int i = 2; i < M; i++) {
        har[i] = (har[i-1] + modInverse(i,MOD))%MOD;
    }
}


int main() {
	pre();
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    vector<ll> v(n+1);
	    for (int i = 1; i <= n; i++) {
	        cin>>v[i];
	    }
	    ll e = har[n-1];
	    ll ans = (e*v[1])%MOD;
	    for (int i = 2; i <= n; i++) {
	        e = e + modInverse(i-1,MOD);
	        e %= MOD;
	        e = (e-modInverse(n-i+1,MOD)+MOD)%MOD;
	        ans += (e*v[i])%MOD;
	        ans %= MOD;
	    }
	    cout<<ans<<endl;
	    
	}

}
