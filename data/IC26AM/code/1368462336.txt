const int N = 200005;

    H[0] = 0;
    for(int i = 1; i < N; i++) H[i] = (H[i - 1] + inv[i]) % mod;
	// your code goes here
	int t;

	while(t--){
	    int n;
	    cin>>n;

	}

}
