#include <bits/stdc++.h>
using namespace std;
#define int long long

const int mod = 998244353;

int power(int a,int b){
    int res = 1;
    while(b>0){
        if(b&1ll)res = (res*a)%mod;
        b = b/2;
        a = (a*a)%mod;
    }
    return res;
}

int inv_mod(int a){
    return power(a,mod-2);
}

int32_t main() {
	// your code goes here
    int t;
    cin>>t;
    
    int mx = 2*1e5+1;
    
    vector<int>h(mx+1);
    
    h[0] = 0;
    
    for(int i=1;i<=mx;i++){
        h[i] = (h[i-1]+ inv_mod(i))%mod;
    }
    
    
    while(t--){
        int n;
        cin>>n;
        
        vector<int>v1(n+1);
        for(int i=1;i<=n;i++)cin>>v1[i];
        
        int ans = 0;
        
        for(int i=1;i<=n;i++){
            ans = (ans + ((v1[i]%mod)*((h[i-1]+h[n-i])%mod))%mod)%mod;
        }
        cout<<ans<<endl;
    }
}
