import sys
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    
    MOD = 998244353
    MAXN = 200005
    
    inv = [0] * MAXN
    if MAXN > 1:
        inv[1] = 1
    for i in range(2, MAXN):
        inv[i] = (MOD - MOD // i) * inv[MOD % i] % MOD
    
    H = [0] * MAXN
    for i in range(1 , MAXN):
        H[i] = (H[i-1] + inv[i]) % MOD
        
    t = int(input_data[0])
    idx = 1
    output = []
    
    for _ in range(t):
        n = int(input_data[idx])
        idx += 1 
        
        ans = 0 
        for i in range(n):
            a_i = int(input_data[idx])
            idx += 1 
            
            coef = (H[i] + H[n - 1 - i]) % MOD
            ans = (ans + a_i * coef) % MOD
            
        output.append(str(ans))
        
    print('\n'.join(output))
    
if __name__ == '__main__':
    solve()
