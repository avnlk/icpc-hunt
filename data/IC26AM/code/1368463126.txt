MOD = 998244353
MAXN = 200000

inv = [0] * (MAXN + 1)
inv[1] = 1

for i in range(2, MAXN + 1):
    inv[i] = MOD - (MOD // i) * inv[MOD % i] % MOD
H = [0] * (MAXN + 1)

for i in range(1, MAXN + 1):
    H[i] = (H[i - 1] + inv[i]) % MOD

t = int(input())

for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    ans = 0 
    
    for i in range(n):
        coefficient = (H[i] + H[n - i - 1]) % MOD
        
        ans = (ans + a[i] * coefficient) % MOD
        
    print(ans)
