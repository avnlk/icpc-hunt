import sys
MOD = 998244353
MAXN = 200005

inv = [0]*MAXN
inv[1]=1
for i in range(2,MAXN):
    inv[i] = MOD-(MOD//i)*inv[MOD%i]%MOD
    
H = [0]*MAXN
for i in range(1, MAXN):
        H[i] = (H[i-1]+inv[i])%MOD
def solve():
    input_data = sys.stdin.read().split()
    if not input_data:
        return
    t = int(input_data[0])
    idx = 1 
    out = []
    
    for _ in range(t):
        n = int(input_data[idx])
        idx += 1 
        a = list(map(int,input_data[idx:idx+n]))
        idx += n 
        ans = 0 
        
        for i in range(n):
            ans = (ans + a [i] * (H[i]+H[n-1-i]))%MOD
        out.append(str(ans))
            
    sys.stdout.write("\n".join(out)+"\n")
if __name__=='__main__':
    solve()
