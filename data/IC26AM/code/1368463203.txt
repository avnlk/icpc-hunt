#include <bits/stdc++.h>
using namespace std;
#define ll long long
const ll mod = 998244353;

ll binpow(ll a, ll b) {
    ll res = 1;
    while(b) {
        if(b & 1)
            res = (res * a) % mod;
            
        a = (a * a) % mod;
        b /= 2;
    }
    return res;
}

void solve() {
    ll n;
    cin >> n;
    vector<ll> a(n);
    for(int i = 0; i < n; i++) cin >> a[i];
    vector<ll> dd(n + 1);
    dd[1] = 1;
    for(int i = 2; i <= n; i++) {
        dd[i] = dd[i - 1];
        dd[i] = (dd[i] + binpow(i, mod - 2)) % mod;
    }
    
    ll ans = 0;
    for(int i = 0; i < n; i++) {
        ll l = i, r = n - 1 - i;
        ll val = (dd[l] + dd[r]) % mod;
        val = (val * a[i]) % mod;
        ans = (ans + val) % mod;
    }
    
    cout << ans << endl;
}

int main() {
	// your code goes here
    int t;
    cin >> t;
    while(t--) solve();
}
