#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;

int main() {
    const int N = 200005;
    vector<long long> inv(N), H(N,0);
    inv[1] = 1;
    for(int i=2; i<N; i++){
        inv[i] = (MOD - (MOD / i) * inv[MOD % i] % MOD) % MOD;
    }
    for(int i=1; i<N; i++){
        H[i] = (H[i-1] + inv[i]) %MOD;
    }
    
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    long long ans = 0;
	    for(int i=1; i<=n; i++){
	        long long a;
	        cin>>a;
	        a %= MOD;
	        long long c = (H[i-1] + H[n-i]) % MOD;
	        ans = (ans + a * c) %MOD;
	    }
	    cout<<ans<<"\n";
	}
	
	return 0;

}
