import sys 
input=sys.stdin.readline 
mod=998244353
maxn=20000
inv=[0]*(maxn+1)
harami=[0]*(maxn+1)
inv[1]=1
for i in range(1,maxn+1):
    if i>1:
        inv[i]=mod-(mod//i)*inv[mod%i]%mod
    harami[i]=(harami[i-1]+inv[i])%mod
t=int(input())
for _ in range(t):
    n=int(input())
    a=list(map(int,input().split()))
    res=0 
    for i in range(n):
        coef=(harami[i]+harami[n-1-i])%mod
        res=(res+a[i]*coef)%mod
    print(res)
