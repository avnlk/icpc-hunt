# cook your dish here
M = 998244353 
N = 200005
inv = [0,1] + [0] * (N - 1)
for i in range(2,N+1):
    inv[i] = (M - (M//i) * inv[M%i]%M)%M 
h = [0] * (N+1)
for i in range(1,N+1):
    h[i] = (h[i-1]+inv[i]%M)
t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int,input().split()))
    ans = 0 
    for j in range(n):
        ans += a[j] * (h[j] + h[n-1-j])
    print(ans%M)
