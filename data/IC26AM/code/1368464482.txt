import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
        long MOD = 998244353L;
        int MAX = 200000;
        
        long[] inv = new long[MAX+1];
        long[] h = new long[MAX +1];
        
        inv[1]=1;
        for(int i=2;i<=MAX;i++){
            inv[i]= MOD -(MOD/i)*inv[(int)(MOD%i)]%MOD;
        }
        for(int i =1;i<=MAX;i++){
            h[i]=(h[i-1]+inv[i])%MOD;
        }
        Scanner sc = new Scanner (System.in);
        int t= sc.nextInt();
        
        while(t-->0){
            int n = sc.nextInt();
            long ans =0;
            for(int i=0;i<n;i++){
                long x = sc.nextLong();
                long coefficient = (h[i]+h[n-1-i])%MOD;
                ans = (ans+(x%MOD)*coefficient)%MOD;
            }
            System.out.println(ans);
        }
	}
	
}
