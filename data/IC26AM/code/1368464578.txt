#include <bits/stdc++.h>
using namespace std;

using ll = long long;

const ll MOD =  998244353;


int main() {
	// your code goes here
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int t;
    cin >> t;
    while(t--) {
        int n;
        cin >> n;
        
        vector<ll> a(n), inv(n+1), h(n+1);
        
        for(auto& x : a) {
            cin >> x;
        }
        
        if(n >= 1) inv[1]=1;
        
        for(int i = 2; i <= n; i++) {
            inv[i] = MOD - (MOD / i) * inv[MOD % i] % MOD;
        }
        
        for(int i = 1; i <= n; i++) {
            h[i] = (h[i-1] + inv[i]) % MOD;
        }
        
        ll ans = 0;
        
        for(int i = 0; i < n; i++) {
            ans = (ans + a[i] * (h[i] + h[n - 1 - i])) % MOD;
        }
        
        cout << ans << '\n';
    }
}
