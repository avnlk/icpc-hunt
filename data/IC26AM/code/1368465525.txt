#include <bits/stdc++.h>
using namespace std;

#define ll long long 
#define vi vector<ll>
#define vvi vector<vi>
#define pii pair<ll,ll>

const ll inf = 1e18; 
const ll mod = 998244353; 

ll pw(ll x, ll y){
    if(x == 0){
        if(y == 0) return 1;
        else return 0; 
    }
    ll res = 1; 
    while(y){
        if(y&1)res = (res * x)%mod; 
        x = (x * x)%mod; 
        y = y /2; 
    }
    return res; 
}

ll inv(ll x){
    return pw(x, mod - 2); 
}

void solve(){
    ll n; cin >> n; 
    vi a(n + 2); 
    for(ll i=1; i<=n; i++) cin >> a[i]; 
    vi h(n + 1, 0); 
    for(ll i=1 ; i<=n; i++)h[i] = (h[i-1] + inv(i))%mod; 
    
    ll ans = 0;
    for(int i=1;i<n;i++) 
    {
        ans += a[i] * (h[n-i]);
        ans %= mod;
    }
    
    reverse(a.begin(),a.end());
    for(int i=1;i<n;i++) 
    {
        ans += a[i] * (h[n-i]);
        ans %= mod;
    }
    cout << ans << "\n";
}



int main() {
	// your code goes here
    ll tc; cin >> tc; 
    while(tc--)solve(); 
}
