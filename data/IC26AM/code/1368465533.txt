import sys
MOD=998244353
MAX=200005
inve=[0]*(MAX+1)
inve[1]=1
for h in range(2,MAX+1):
    inve[h]=(MOD-(MOD//h)*inve[MOD%h]%MOD)%MOD
J=[0]*(MAX+1)
for h in range(1,MAX+1):
    J[h]=(J[h-1]+inve[h])%MOD
tc=int(input())
result=[]
for h in range(tc):
    nm=int(input())
    arr=list(map(int,input().split()))
    t=0
    for k in range(1,nm+1):
        ap=arr[k-1]%MOD
        wp=(J[k-1]+J[nm-k])%MOD
        t=(t+ap*wp)%MOD
    result.append(str(t))
sys.stdout.write("\n".join(result))
