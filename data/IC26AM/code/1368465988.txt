import sys
input=sys.stdin.readline

mod=998244353

maxi=200000

inv=[0]*(maxi+1)
inv[1]=1

for i in range(2,maxi+1):
    inv[i]=mod-(mod//i)*inv[mod%i]%mod 
    
h=[0]*(maxi+1)

for i in range(1,maxi+1):
    h[i]=(h[i-1]+inv[i])%mod 

t=int(input())

for _ in range(t):
    n=int(input())
    a=list(map(int,input().split()))
    
    ans=0 
    
    for i in range(n):
        coeff=(h[i]+h[n-1-i])%mod 
        ans=(ans+(a[i]%mod)*coeff)%mod 
    print(ans)
