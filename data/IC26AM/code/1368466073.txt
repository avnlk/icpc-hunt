#include<stdio.h>
#include<stdlib.h>

#define MOD 998244353LL

int main(){
    int t;
    scanf("%d", &t);
    
    while(t--){
        int n;
        scanf("%d", &n);
        long long *a = malloc(n * sizeof(long long));
        long long *h = calloc(n, sizeof(long long));
        long long *inv = calloc(n, sizeof(long long));
        
        for(int i=0;i<n;i++)
            scanf("%lld", &a[i]);
        if(n>1){
            inv[1]=1;
            for (int i = 2;i<n;i++)
                inv[i] = MOD-(MOD/i) * inv[MOD % i] % MOD;
            for (int i = 1;i<n;i++)
                h[i] = (h[i-1]+inv[i]) % MOD;
        }
        long long ans = 0;
        
        for( int i=0;i<n;i++){
            long long coef = (h[i]+h[n-i-1]) % MOD;
            ans = (ans + a[i] * coef) % MOD;
        }
        
        printf("%lld\n", ans);
        free(a);
        free(h);
        free(inv);
    }
    return 0;
}
