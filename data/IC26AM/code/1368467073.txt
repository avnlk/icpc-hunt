T = int(input())

MOD = 998244353

for t in range(T):
    
    n = int(input())
    a = list(map(int, input().split()))
    
    h = [0]*(n+1)
    
    for i in range(1, n+1):
        h[i]=(h[i-1] + pow(i, MOD-2, MOD)) % MOD
        
    ans = 0
        
    for i in range(n):
        w = (h[i] + h[n-i-1]) % MOD
        ans = (ans + a[i]*w) % MOD
        
    print(ans)
