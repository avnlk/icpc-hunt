const ll MOD=998244353;
long long binpow(ll b,ll a)
{
    ll r=1;
    b=b%MOD;
    while(a>0)
    {
        if(a&1)r=r*b%MOD;
        b=b*b%MOD;
        a=a/2;
    }
    return r;
}

int main() {
	// your code goes here
    int t; cin >> t;
    while(t--){
        ll n; cin >> n;
        vector<ll> a(n+1);
        for(int i=1;i<=n;i++)cin>>a[i];
        
        vector<ll>pre(n+1,0);
        for(int i=1;i<=n;i++)
        {
            pre[i]=(pre[i-1]%MOD + a[i]%MOD)%MOD;
        }
        
        ll ans=0;
        for(int i=1;i<=n-1;i++)
        {
            
            ll suff=(pre[n]-pre[i]+MOD)%MOD;
            ll sum=(suff%MOD+pre[n-i]%MOD)%MOD;
            ans=(ans+sum*binpow(i,MOD-2))%MOD;
        }
        cout<<ans<<endl;
        
    }
}
