#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() 
{
    ll M = 998244353;
    int t;
    cin >> t;
    vector<vector<ll>>a(t);
    int k = 1;
    for(int j=0;j<t;j++)
    {
        int n;
        cin >> n;
        a[j].push_back(0);
        for(int i=1;i<=n;i++)
        {
            ll x;
            cin >> x;
            a[j].push_back(x);
        }
        k = max(k,n);
    }
    vector<ll>v(k+2),h(k+2);
    v[1]=1;
    for(int i=2;i<=k;i++)
    {
        v[i]=(M-M/i)*v[M%i]%M;
    }
    h[0]=0;
    for(int i=1;i<=k;i++)
    {
        h[i]=(h[i-1]+v[i])%M;
    }
    for(int j=0;j<t;j++)
    {
        int n = a[j].size()-1;
        ll r = 0;
        for(int i=1;i<=n;i++)
        {
            ll d = a[j][i]%M * ((h[i-1]%M + h[n-i])%M)%M;
            r = (r+d)%M;
        }
        cout << r << "\n";
        }
        return 0;
}
