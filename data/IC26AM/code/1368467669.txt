import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
    static final int MOD = 998244353;
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc = new Scanner(System.in);
		
		if(!sc.hasNextInt()) return;
		
		int Max = 200005;
		
		long[] inv = new long[Max];
		
		inv[1] = 1;
		
		for(int i = 2; i < Max; i++){
		    inv[i] = (MOD -(MOD/i)* inv[MOD % i] % MOD) % MOD;
		    
		}
		long[] h = new long[Max];
		
		h[0] = 0;
		
		for(int i = 1; i < Max; i++){
		    h[i] = (h[i-1] + inv[i]) % MOD;
		}
		int t = sc.nextInt();
		
		while(t --> 0){
		    int n = sc.nextInt();
		    long[] a = new long[n];
		    
		    for(int i = 0; i < n; i++){
		        a[i] = sc.nextLong();
		    }
		    
		    if(n == 1){
		        System.out.println(0);
		        continue;
		    }
		    
		    long Ts = 0;
		    
		    for(int i = 0; i < n; i++){
		        long coffe = (h[i] + h[n-1-i]) % MOD;
		        long contri = (a[i] % MOD) * coffe % MOD;
		        Ts = (Ts + contri) % MOD;
		    }
		    System.out.println(Ts);
		}

	}
}
