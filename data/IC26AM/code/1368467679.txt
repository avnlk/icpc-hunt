import sys# cook your dish here
MOD = 998244353
MAXN = 200000
H = [0]*(MAXN + 1)
inv = [0] *(MAXN + 1)
inv[1] = 1
for i in range (2,MAXN+1):
    inv[i] = MOD-(MOD // i) * inv[MOD % i ]% MOD
for i in range(1,MAXN+1):
    H[i] = (H[i-1] + inv[i]) % MOD
data = list(map(int,sys.stdin.buffer.read().split()))
t = data[0]
pos = 1 
out = []
for _ in range(t):
    n = data[pos]
    pos += 1 
    a = data[pos:pos+n]
    pos += n
    ans = 0 
    for i in range(n):
        coff = H[i] + H[n-1-i]
        ans = (ans + a[i]*coff)%MOD
    out.append(str(ans))
sys.stdout.write("\n".join(out))
