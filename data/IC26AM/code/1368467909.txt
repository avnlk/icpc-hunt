#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 998244353;
const int N =200005;
ll inv[N],h[N];

int main() {
	// your code goes here
	int t;
	cin>>t;
	inv[1] = 1;
	for(int i=2;i<N;i++) inv[i] = (MOD-(MOD/i)*inv[MOD%i]%MOD)%MOD;
	h[0] =0;
	for(int i=1;i<N;i++)h[i] = (h[i-1]+inv[i])%MOD;
	
	
	while(t--){
	    int n;
	    cin>>n;
	    ll ans =0;
	    for(int i=1;i<=n;i++){
	        ll x ; cin>>x;
	        ll c =(h[i-1] +h[n-i])%MOD;
	        ans =(ans + x%MOD*c)%MOD;
	    }
	    cout<<ans<<"\n";
	    
	    
	}
	
	return 0;

}
