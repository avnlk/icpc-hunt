#include <bits/stdc++.h>
using namespace std;
using ll=long long;
const ll MOD=998244353;

ll binary_expo(ll a,ll b){
      ll ansti=1;
      while(b>0){
           if(b%2){
               ansti=ansti*a%MOD;
           }
           a=a*a%MOD;
           b=b/2;
      }
      return ansti;
}

int main() {
	ll t;
	cin>>t;
	
	while(t--){
	     ll n;
	     cin>>n;
	     
	     vector<ll>a(n);
	     for(ll i=0;i<n;i++) cin>>a[i];
	     
	     
	     vector<ll>h(n,0);
	     
	     for(ll i=1;i<n;i++){
	          int inv=binary_expo(i,MOD-2);
	          
	          h[i]=(h[i-1]+inv)%MOD;
	     }
	     
	     ll ansi=0;
	     for(ll i=0;i<n;i++){
	         ll contri=(h[i]+h[n-1-i])%MOD;
	         ansi=(ansi+a[i]*contri)%MOD;
	     }
	     
	     cout<<ansi<<endl;
	    
	}
}
