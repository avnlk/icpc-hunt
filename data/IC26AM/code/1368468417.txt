import sys

MOD = 998244353
MAXN = 200000

inv = [0] * (MAXN +1)
inv[1] = 1

for i in range(2, MAXN + 1):
    inv[i] = MOD - (MOD // i) * inv[MOD % i] %MOD
    
H = [0] * (MAXN + 1)
    
for i in range(1, MAXN + 1):
  H[i] = (H[i - 1] + inv[i]) % MOD
 
def solve():
  data = list(map(int,
sys.stdin.buffer.read().split()))
  it = iter(data)
  
  t = next(it)
  ans = []
  
  for _ in range(t):
    n = next(it)
    
    a = [next(it) for _ in range(n)]
    
    total = 0
    
    for i in range(n):
        
      coeff = (H[i] + H[n - i - 1]) % MOD
      total = (total + a[i] * coeff) % MOD
      
    ans.append(str(total))
    
  print("\n".join(ans))

solve()
