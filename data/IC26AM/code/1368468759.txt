#include <bits/stdc++.h>
using namespace std;

const long long MOD = 998244353;
const int MAXN = 200005;
long long inv_[MAXN],H[MAXN];

int main() {
	// your code goes here
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    inv_[1] = H[1] = 1;
    
    for(int i = 2; i < MAXN; i++){
        inv_[i] = (MOD- MOD / i) * inv_[MOD % i] % MOD;
        H[i] = (H[i-1] + inv_[i]) % MOD;
    }
    
    int t;
    cin >> t;
    while(t--){
        int n ;
        cin >> n;
        long long ans = 0, x;
        for(int i = 1; i <= n; i++){
            cin >> x;
            ans = (ans + (H[i - 1] + H[n -i] ) % MOD * x) % MOD;
        }
        cout << ans << "\n";
    }
}
