#include <stdio.h>

#define MOD  998244353
long long h[200005], inv[200005], a[200005];

int main() {
    int t,n,i;
    long long ans,coef;
    
    inv[1]=1;
    for(i=2;i<=200000;i++)
        inv[i]=MOD-(MOD/i)*inv[MOD%i]%MOD;
    
    for(i=1;i<=200000;i++)
        h[i]=(h[i-1]+inv[i])%MOD;
        
    scanf("%d",&t);
    
    while(t--) {
        scanf("%d",&n);
        
        for(i=0;i<n;i++)
            scanf("%d",&a[i]);
            
        ans=0;
        
        for(i=0;i<n;i++){
            coef=(h[i]+h[n-i-1])%MOD;
            ans=(ans+a[i]*coef)%MOD;
        }    
        printf("%d\n",ans);
    }
    return 0;
}
