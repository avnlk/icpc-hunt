#include <bits/stdc++.h>
using namespace std;
const long long MOD=998244353;

int main() {
	// your code goes here
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	const int maxn=200000;
	vector<long long>inv(maxn+1);
	inv[1]=1;
	for(int i=2;i<=maxn;i++){
	    inv[i]=MOD-(MOD/i)*inv[MOD%i]%MOD;
	    
	}
	vector<long long>h(maxn+1,0);
	for(int i=1;i<=maxn;i++){
	    h[i]=(h[i-1]+inv[i])%MOD;
	}
	while(t--){
	    int n;
	    cin>>n;
	    vector<long long> a(n+1);
	    for(int i=1;i<=n;i++){
	        cin>>a[i];
	    }
	    long long ans=0;
	    for(int i=1;i<=n;i++){
	        long long coff=(h[i-1]+h[n-i])%MOD;
	        ans=(ans+a[i]*coff)%MOD;
	    }
	    cout<<ans<<endl;
	}
	return 0;
}
