#include <bits/stdc++.h>
using namespace std;
const int mod = 998244353;
const int maxn = 200000;
int main() {
    
    std::vector<long long> inv(maxn+1), h(maxn+1);
    inv[1] = 1;
    for(int i= 2;i<=maxn;i++){
        inv[i] = mod - (mod/i) *inv[mod % i] % mod;
    }
    h[0] = 0;
    for(int i = 1;i<=maxn;i++){
        h[i] = (h[i-1] + inv[i]) % mod;
    }
    
    int t;
    cin >> t;
    while(t--){
        int n;
        cin >> n;
        vector<long long >a(n);
        for(int i = 0;i<n;i++){
            cin >> a[i];
        }
        long long ans = 0;
        for(int i = 0;i<n;i++){
            long long co = (h[i] + h[n-1-i]) % mod;
            ans = (ans + a[i] * co) % mod;
        }
        cout << ans << endl;
    }
	return 0;
}
