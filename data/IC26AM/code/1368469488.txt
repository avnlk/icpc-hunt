#include <bits/stdc++.h>
using namespace std;

const long long MOD = 998244353;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while(T--){
        int n;
        cin >> n;
        vector <long long>a(n);
        for (auto &x : a){
            cin >> x;
        }
        vector<long long> inv(n + 1);
        inv[1] = 1;
        for (int i =2; i <= n; i++){
            inv[i] = MOD - (MOD / i)*inv[MOD % i] % MOD;
        }
        vector<long long> H(n + 1, 0);
        for (int i = 1; i <= n; i++){
            H[i] = (H[i-1] + inv[i]) % MOD;
        }
        long long ans;
        for(int i = 0; i < n; i++){
            long long coef = (H[i] + H[n - 1 - i]) % MOD;
            ans = (ans + a[i] * coef) % MOD;
        }
        cout << ans << "\n";
    }
    return 0;

}
