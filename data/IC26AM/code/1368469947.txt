#include <bits/stdc++.h>
using namespace std;
#define int long long

const int MOD = 998244353;

signed main() {
	// your code goes here
	int t;
	cin>>t;
	while(t--) {
	    int n;
	    cin>>n;
	    
	    vector<int> a(n + 1);
	    for(int i = 1; i<=n; i++) {
	        cin>>a[i];
	        a[i] %= MOD;
	        if(a[i] < 0) a[i] += MOD;
	    }
	    
	    vector<int> inv(n + 1);
	    vector<int> H(n +1);
	    
	    if(n >= 1) inv[1] = 1;
	    
	    for(int i = 2; i<=n; i++) {
	        inv[i] = MOD - (MOD/i) * inv[MOD % i] % MOD;
	    }
	    for(int i = 1; i<=n;i++) {
	        H[i] = (H[i - 1] + inv[i]) % MOD;
	    }
	    int ans = 0;
	    for(int i = 1; i<=n; i++) {
	        int coefficient = (H[i - 1] + H[n - i]) % MOD;
	        ans = (ans + a[i] * coefficient) % MOD;
	    }
	    cout<<ans<<"\n";
	    
	    
	}

}
