#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;
const int MAXN = 200005;

long long inv[MAXN];
long long H[MAXN];

void precompute(){
    inv[1]  =1;
    for(int i=2; i< MAXN; i++){
        inv[i] = (MOD- MOD/i) * inv[MOD% i] % MOD;
    }
    
    H[0] = 0;
    for(int i=1; i< MAXN; i++){
        H[i] = (H[i-1] + inv[i])%MOD;
    }
}

void solve(){
    int n;
    cin>>n;
    vector<long long>a(n);
    for(int i=0; i< n; i++) cin>>a[i];
    
    if( n== 1){
         cout<<0<<endl;
         return ;
    }
    long long ans = 0;
    for(int i=1; i <= n; i++){
        long long c_i = (H[i-1] + H[n-i]) %  MOD;
        ans = (ans + (a[i-1] % MOD) * c_i) % MOD;
    }
    cout<<ans<<endl;
    
}

int main() {
	// your code goes here
	
	precompute();
	int t;
	cin>>t;
	
	while(t--){
	    solve();
	}
	return 0;
}
