#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const ll MOD = 998244353;
int main() {
	int T;
	cin>> T;
	vector<int> ns(T);
	vector<vector<ll>> arr(T);
	
	int mx = 0;
	
	for(int tc = 0; tc < T; tc++){
	    int n ;
	    cin >> n;
	    ns[tc] = n;
	    mx = max(mx, n);
	    arr[tc].resize(n);
	    for(auto &x :  arr[tc])
	    cin >> x;
	}
	
	vector<ll> inv(mx + 1);
	if(mx >= 1)
	inv[1] = 1;
	
	for(int i = 2; i <= mx; i++){
	    inv[i] = MOD - (MOD / i) * inv[MOD % i] % MOD;
	}
	vector<ll> H(mx + 1, 0);
	for(int i = 1; i <= mx; i++){
	    H[i] = (H[i - 1] + inv[i]) % MOD;
	}
	
	for(int tc = 0; tc < T; tc++){
	    int n = ns[tc];
	    ll ans = 0;
	    
	    for(int i = 0; i < n; i++){
	        ll coeff = (H[i] + H[n - i - 1]) % MOD;
	        
	        ans = (ans + (arr[tc][i] % MOD) * coeff) % MOD;
	    }
	    cout << ans << '\n';
	}
	return 0;

}
