import sys
input = sys . stdin.readline
MOD=998244353
N=200000
inv=[0]*(N+1)
h=[0]*(N+1)
inv[1]=1
for i in range(2,N+1):
    inv[i]=MOD - (MOD//i)* inv[MOD%i]%MOD
for i in range(1,N+1):
    h[i]=(h[i-1]+inv[i])%MOD
for i in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    ans=0 
    for i ,x in enumerate(a):
        ans += x*(h[i]+h[n-1-i])
    print(ans%MOD)
