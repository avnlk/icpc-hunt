#include <bits/stdc++.h>
using namespace std;

#define int long long

const int MOD = 998244353;

int bp(int x, int p)
{
    if(p==0) return 1;
    int q = bp(x, p/2);
    q *= q;
    q %= MOD;
    if(p%2)
    {
        q *= x;
        q %= MOD;
    }
    return q;
}

int mi(int x)
{
    x %= MOD;
    return bp(x, MOD-2);
}

const int MXN = 200005;

int fact[MXN]{};
int inv[MXN]{};

int nCr(int n, int r)
{
    int ans = fact[n];
    ans *= inv[r];
    ans %= MOD;
    ans *= inv[n-r];
    ans %= MOD;
    return ans;
}

void solve()
{
    int n;
    cin>>n;
    
    vector<int> a(n+1, 0);
    
    for(int i=1; i<=n; i++)
    {
        cin>>a[i];
    }
    
    vector<int> pref(n+1, 0);
    for(int i=1; i<=n; i++)
    {
        pref[i] = pref[i-1] + mi(i);
        pref[i] %= MOD;
    }
    
    int ans = 0;
    for(int i=1; i<=n; i++)
    {
        int xx = pref[i-1] + pref[n-i];
        xx %= MOD;
        ans += xx*a[i];
        ans %= MOD;
    }
    
    cout<<ans<<"\n";
}

signed main() {
	// your code goes here

	cin.tie(0);
	
	fact[0] = 1;
	inv[0] = 1;
	for(int i=1; i<MXN; i++)
	{
	    fact[i] = fact[i-1]*i;
	    fact[i] %= MOD;
	    inv[i] = mi(fact[i]);
	}
	
	int t = 1;
	cin>>t;
	while(t--)
	{
	    solve();
	}
}
