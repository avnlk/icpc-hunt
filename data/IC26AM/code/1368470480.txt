#include <bits/stdc++.h>
#define mod 998244353
using namespace std;

vector<long long> inv_,H;
void solve(int n){
    if (inv_.empty()){
        inv_.assign(2,1);
        H.assign(2,0);
        H[1]=1;
        int cur=inv_.size()-1;
        if(n<=cur) return;
        inv_.resize(n+1);
        H.resize(n+1);
        for(int i=cur+1; i<=n; i++){
            inv_[i]=(mod-(mod/i)*inv_[mod%i]%mod)%mod;
            H[i]=(H[i-1]+inv_[i])%mod;
            
        }
    }
}

int main() {
	// your code goes here
    int t;
    cin>>t;
    while(t--){
        int n;
        cin>>n;
        solve(n);
        long long ans = 0;
        long long temp;
        for(int j=1;j<=n;j++){
            long long a;
            cin>>a;
            a%=mod;
            if(a<0) a+=mod;
            long long w=(H[j-1]+H[n-j])%mod;
            ans=(ans+a*w)%mod;
        }
        cout<<ans<<endl;
    }
    return 0;
}
