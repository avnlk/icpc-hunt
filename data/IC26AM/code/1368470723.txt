#include <bits/stdc++.h>
using namespace std;
using int64= long long;
static const int64 mod=998244353;
static const int maxn = 200000;

int64 inv[maxn+1];
int64 h[maxn+1];

int main() {
	// your code goes here
	
	
	
	inv[1]=1;
	for(int i=2;i<=maxn;++i){
	    inv[i]=mod-(mod/i)*inv[mod%i]%mod;
	    
	    
	    
	    
	}
	h[0]=0;
	for(int i=1;i<=maxn;++i){
	    h[i]=(h[i-1]+inv[i])%mod;
	}
	int t;
	cin>>t;
	
	while(t--){
	    int n;
	    cin>>n;
	    
	    int64 ans=0;
	    
	    for(int j=1;j<=n;++j){
	        int64 a;
	        cin>>a;
	        
	        
	        
	        int64 coeff =(h[j-1]+h[n-j])%mod;
	        
	        ans=(ans+(a%mod)*coeff)%mod;
	        
	        
	    }
	    cout<<ans<<endl;
	}
	return 0;

}
