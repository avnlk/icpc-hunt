#include <bits/stdc++.h>
using namespace std;
const long long MOD = 998244353;
const int MAXN = 200005;
long long inv_[MAXN],H[MAXN];

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    inv_[1] = 1;
    for(int i = 2;i<MAXN;i++)
    inv_[i] = (MOD-(MOD/i)*inv_[MOD%i]%MOD)%MOD;
    H[0] = 0;
    for(int i = 1;i<MAXN;i++)
    H[i] = (H[i-1]+inv_[i])%MOD;
    int t;
    cin>>t;
    while(t--)
    {
        int n;
        cin >>n;
        long long ans = 0;
        for(int j = 1;j<=n;j++)
        {
            long long a;
            cin >> a;
            long long d = (H[j-1] + H[n-j]) % MOD;
            ans = (ans + a%MOD*d)%MOD;
        }
        cout << ans << '\n';
        
    }
    return 0;

}
