import sys
def solve():
    input=sys.stdin.read().split()
    if not input:
        return
    mod=998244353
    max_n=200005
    inv=[0]*max_n
    if max_n>1:
        inv[1]=1
    for i in range(2,max_n):
        inv[i]=(mod-mod//i)*inv[mod%i]%mod
    h=[0]*max_n
    for i in range(1,max_n):
        h[i]=(h[i-1]+inv[i])%mod
    iterator=iter(input)
    t=int(next(iterator))
    out=[]
    for _ in range(t):
        n=int(next(iterator))
        a=[int(next(iterator)) for _ in range(n)]
        if n<=1:
            out.append("0")
            continue
        ans=0
        for i in range(n):
            coef=(h[i]+h[n-1-i])%mod
            ans=(ans+a[i]*coef)%mod
        out.append(str(ans))
    sys.stdout.write('\n'.join(out)+'\n')
if __name__=='__main__':
    solve()
