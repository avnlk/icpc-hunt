# cook your dish here
MOD= 998244353
t= int(input())
tests=[]
for _ in range(t):
    n=  int(input())
    a= list(map(int, input().split()))
    tests.append((n,a))
max_n = max(n for n,a in tests)
inv= [0]*(max_n+1)
if max_n >=1:
    inv[1]=1
for i in range(2,max_n+1):
    inv[i]= MOD- (MOD//i)*inv[MOD%i]%MOD
H=[0]*(max_n+1)
for i in range(1, max_n +1):
    H[i]= (H[i-1]+inv[i])%MOD
for n,a in tests:
    ans=0
    for i in range(n):
        coeff= (H[i]+H[n-1-i])%MOD
        ans= (ans+ a[i]*coeff)%MOD
    print(ans)
