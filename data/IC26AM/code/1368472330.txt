#include <bits/stdc++.h>
using namespace std;

#define ll long long

const ll mod = 998244353;
const int n = 200000;

ll inv[n], h[n];


void func() {
    inv[1] = 1;
    
    for(int i = 2; i < n; i++) {
        inv[i] = mod - (mod / i) * inv[mod % i] % mod;
    }
    
    for(int i = 1; i < n; i++) {
        h[i] = (h[i-1] + inv[i]) % mod;
    }
}

int main() {
	// your code goes here
    int t;
    cin >> t;
    func();
    while(t--) {
        int n; cin >> n;
        vector<ll>arr(n);
        for(auto &i : arr) cin >> i;
        
        ll ans = 0;
        for(int i = 0; i < n; i++) {
            ll x = h[i] + h[n-i-1];
            x = x % mod; 
            
            ans = (ans%mod +( arr[i] * x) %mod ) %mod;
        }
        cout << ans << "\n";
    }
}
