m = 998244353
fact=[1]*int(2e5+1)
for i in range(1,int(2e5+1)):
    fact[i]=(fact[i-1]*i)%m
a1l = [0]*int(2e5+1)
for i in range(2,int(2e5+1)):
    a1l[i] = ((i-1)*a1l[i-1] + fact[i-2]) % m

m = 998244353
for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    a1 = a1l[n] 
    if n==1:
        print(0)
        continue
    nf = ((n-2) * fact[n-2])%m
    f = fact[n-1]
    ans = 0
    if n%2:
        for i in range(n//2):
            ans = (ans + (a[i] + a[n-1-i])*a1) % m
            a1 = (a1 + nf) % m
        ans = (ans + a[n//2]*a1) % m
    else:
        for i in range(n//2):
            ans = (ans + (a[i] +a[n-1-i])*a1) % m
            a1 = (a1 + nf) % m
    fa = (ans * pow(f,-1,m)) % m
    print(fa)
