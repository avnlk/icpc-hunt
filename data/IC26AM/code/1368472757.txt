import sys

def solution():
    i_data = sys.stdin.read().split()
    if not i_data:
        return
    
    MOD = 998244353
    maxi = 200005
    
    inv = [0]*maxi
    
    if maxi > 1:
        inv[1] = 1
    for i in range(2, maxi):
        inv[i] = (MOD-MOD//i)*inv[MOD%i]%MOD
        
    h = [0]*maxi
    for i in range(1, maxi):
        h[i] = (h[i-1]+inv[i])%MOD
    
    t = int(i_data[0])
    index = 1 
    
    out = []
    
    for _ in range(t):
        n = int(i_data[index])
        index += 1 
        
        a = i_data[index : index+n]
        index += n
        
        e_sum = 0
        
        for i in range(1, n+1):
            count = int(a[i-1])
            cf = (h[i-1]+h[n-i])%MOD
            
            e_sum = (e_sum + count *cf)%MOD 
        out.append(str(e_sum))
        
    sys.stdout.write('\n'.join(out)+'\n')
    
    
if __name__ == '__main__':
    solution()
