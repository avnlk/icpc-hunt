#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

const int M = 998244353;
ll binpow(ll a, ll b){
    if(b==0) return 1ll;
    if(b%2==0){
        ll x = binpow(a,b/2);
        x = (x*x)%M;
        return x;
    }
    else{
        ll x= binpow(a,b/2);
        x = (x*x)%M;
        return (((a%M)*x)%M);
    }
}
void solve() {
	// your code goes here
    int n;
    cin>>n;
    std::vector<ll> v(n) ;
    for(int i = 0; i<n; i++) cin>>v[i];
    if(n==1){
        cout<<0<<endl;
        return;
    }
    
    if(n==2){
        cout<<v[0] + v[1]<<endl;
        return;
    }
    
    ll fact = 1;
    for(ll i = 2; i<=n-1; i++){
        fact = (fact*i)%M;
    }
    
    ll res = 0;
    res = (v[0]*((fact*2)%M -1 + M)%M)%M;
    res = res + (v[n-1]*((fact*2)%M - 1 + M)%M)%M;
    
    for(int i = 1; i<n-1;i++){
        res = (res + (((fact*(n-2))%M + (n-1))%M * v[i])%M)%M;
    }
    ll modinv = binpow(fact,M-2);
    
    res = (res*modinv)%M;
    cout<<res<<endl;    
    
}
int main(){
    int t;
    cin>>t;
    while(t--) {
        solve();
    }
}
