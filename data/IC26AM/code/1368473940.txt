#include <bits/stdc++.h>
using namespace std;
#define ll long long 

long long mod=998244353;


long long binpow(long long b, long long e){
    ll ans=1;
    for(;e;b=b*b%mod,e/=2)
        if (e&1) ans=ans*b % mod;
    return ans%mod;
}

    	fa[1]=1;
    	dp[0]=0;
    	
    	dp[1]=1;
    	for(ll i=2;i<=n;i++){
    	    factp=(factp*i)%mod;
    	    fa[i]=factp%mod;
    	    den=(den+(binpow(i,mod-2)%mod))%mod;
    	    dp[i]=(factp*den)%mod;
    	    
    	}
    	
    	long long ans=0;
    	
    // 	for(int i=0;i<=n;i++){

    	    t1=(t1*fa[n-1])%mod;
    	    t1=(t1*binpow(fa[i],mod-2))%mod;
    	    t1=(t1*binpow(fa[n-i-1],mod-2))%mod;
    	    ans=(ans+t1)%mod;
