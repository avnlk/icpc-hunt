# cook your dish here
MOD=998244353
t=int(input())
for i in range(t):
    n=int(input())
    a=list(map(int,input().split()))
    inv=[0] * (n+1)
    h=[0] * (n+1)
    inv[1]=1
    for i in range(2,n+1):
        inv[i]=MOD-(MOD//i) * inv[MOD % i] % MOD
    for i in range(1,n+1):
        h[i]=(h[i-1]+ inv[i]) % MOD
    ans=0
    for i in range(n):
        ans+=a[i] * (h[i] + h[n-1-i])
    print(ans % MOD)
