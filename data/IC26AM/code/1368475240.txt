import sys
input = sys.stdin.readline
                          
MOD = 998244353
 
t = int(input())
tests = []
mx = 0
for _ in range(t):
    n = int(input())
    a = list(map(int,input().split()))
    tests.append((n,a))
    mx = max(mx,n)
inv = [0 ]*(mx+1)
if mx >=1:
    inv[1]=1
for i in range(2,mx+1):
    inv[i]= MOD - (MOD//i)*inv[MOD % i] % MOD
H = [0]*(mx+1)
for i in range(1, mx+1):
    H[i] = (H[i-1] + inv[i]) %MOD
for n ,a in tests:
    ans=0
    for i in range (n):
        coef = (H[i] + H[n-1-i])%MOD
        ans = (ans + a[i]*coef)%MOD
    print(ans)
