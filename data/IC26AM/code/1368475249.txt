#include <bits/stdc++.h>
using namespace std;
// #define ll long long
typedef long long ll;
// Harmonic ? exp ? 
const int MOD = int(998244353);
ll powexp(ll b, ll e) {
    ll res = 1;
    while(e) {
        if(e & 1) res = 1LL * res * b % MOD;
        b = 1LL * b * b % MOD;
        e /= 2;
    }
    return res;
}
ll modInv(ll a, ll b) {
    return 1LL * a * powexp(b, MOD - 2) % MOD;
}
const int MAXN = 200005;
ll Harm[MAXN + 1];
void pc() {
    Harm[0] = 0LL;
    for(int i = 1; i <= MAXN; i++) {
        Harm[i] = (Harm[i - 1] + modInv(1, i)) % MOD;
    }
}
void solve() {
    int n; cin >> n;
    std::vector<ll> arr(n + 1);
    for(int i = 1; i <= n; i++) cin >> arr[i];
    vector<ll> pref(n + 1, 0);
    for(int i = 1; i <= n; i++) {
        pref[i] = (pref[i - 1] + Harm[i]) % MOD;
    }
    auto range = [&](int l, int r) -> ll {
        if(l > r) return 0;
        
        ll curr = pref[r];
        if(l - 1 >= 1) curr = (curr - pref[l - 1] + MOD) % MOD;
        return curr;
        // return (pref[r] - pref[l - 1] + MOD) % MOD;
    };
    ll ans = 0;
    for(int i = 1; i <= n; i++) {
        /*
        n = 7
        i = 3, 
        (1/1 + 1/2 + 1/3) from left 
        rem = 4
        (1/1 + 1/2 + 1/3 + 1/4) Right
        mm = (1/1 + 1/2 + .. + 1/k)

        */
        int l = i - 1;
        int r = n - i;
        int mm = min(l, r);
        int mx = max(l, r);
        // 2 * (1...mm) + (mm + 1, mx)
        // ans = (ans + 2LL * arr[i] % MOD * pref[mm] % MOD) % MOD;
        // ans = (ans + 1LL * arr[i] * range(mm + 1, mx) % MOD) % MOD;
        ans = (ans + 1LL * arr[i] * Harm[l] % MOD) % MOD;
        ans = (ans + 1LL * arr[i] * Harm[r] % MOD) % MOD;
    }
    cout << ans << "\n";
}

int main() {
