#include <bits/stdc++.h>
using namespace std;

#define int int64_t

const int MOD = 998244353;

int madd(int a, int b)
{
    return (a + b) % MOD;
}

int msub(int a, int b)
{
    return (a - b + MOD) % MOD;
}

int mmul(int a, int b)
{
    return (a * b) % MOD;
}

int mexp(int a, int p)
{
    int val = 1;
    while(p > 0)
    {
        if(p & 1)
            val = mmul(val, a);
        a = mmul(a, a);
        p >>= 1;
    }
    return val;
}

int mdiv(int a, int b)
{
    return mmul(a, mexp(b, MOD - 2));
}


const int N = 3e5+5;
int fac[N], inv[N], ifac[N];

int ncr(int n, int r)
{
    if(n < r or r < 0)
        return 0;
    return mmul(fac[n], mmul(ifac[r], ifac[n - r]));
}

signed main() {
	// your code goes here
	    
	fac[0] = ifac[0] = inv[0] = 1;
	for(int i = 1; i < N; i ++)
	{
	    fac[i] = mmul(fac[i - 1], i);
	    inv[i] = mdiv(1, i);
	    ifac[i] = mmul(fac[i - 1], inv[i]);
	}
	
    int t;
    cin >> t;
    while(t --)
    {
        int n;
        cin >> n;
        
        vector<int> a(n + 1);
        for(int i = 1; i <= n; i ++)
            cin >> a[i];
            
        auto ret = [&]() -> int
        {
            int here = 0;
            
            auto con = [&](int i, int lsum) -> void
            {
                int now = mmul(fac[i - 1], lsum);
                now = mmul(now, a[i]);
                now = mmul(now, fac[n - i]);
                now = mmul(now, ncr(n - 1, i - 1));
                here = madd(here, now);
            };
            
            int lsum = 0;
            for(int i = 2; i <= n; i ++)
            {
                int nadd = mmul(ifac[i - 2], inv[i - 2]);
                lsum = madd(lsum, nadd);
                con(i, lsum);
            }
            
            return here;
        };
        
        int ans = 0;
        ans = madd(ans, ret());
        reverse(a.begin() + 1, a.end());
        ans = madd(ans, ret());

        ans = mmul(ans, ifac[n - 1]);
        
        cout << ans << endl;
    }
}
