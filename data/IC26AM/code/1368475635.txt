#include <bits/stdc++.h>
using namespace std;
const int MOD = 998244353;
const int MAXN= 200005;
long long mopow(long long a, long long b){
    long long res = 1;
    while(b>0){
        if(b&1)res = res *a%MOD;
        a = a*a%MOD;
        b>>=1;
    }
    return res;
    
}
long long inv[MAXN],harm[MAXN];
void precompute(){
    inv[1]=1;
    for(int i =2;i<MAXN;i++){
        inv[i]=MOD -(MOD/i) * inv[MOD%i]%MOD;
    }
    harm[0]=0;
    for (int i =1;i<MAXN;i++){
        harm[i]=(harm[i-1]+inv[i]%MOD);
    }
}
int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	precompute();
	int t;
	cin>>t;
	while(t--){
	    int n;
	    cin>>n;
	    vector<long long>a(n+1);
	    for(int i=1;i<=n;i++){
	        cin>>a[i];
	    }
	    long long ans = 0;
	    for(int i = 1;i<=1;i++){
	        long long coeff=(harm[i-1]+harm[n-i])%MOD;
	        ans =(ans +a[i]*coeff)%MOD;
	    }
	    cout<<ans<<"\n";
	}
	return 0;

}
