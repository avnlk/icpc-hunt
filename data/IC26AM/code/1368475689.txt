#include <bits/stdc++.h>

using namespace std;
using ll = long long;

ll MOD = 998244353;

ll binexp(ll base, ll exp)
{
    ll m = 998244353;
    ll res = 1;
    base %= m;
    while (exp > 0)
    {
        if (exp % 2 == 1)
        {
            res = (res * base) % m;
        }
        base = (base * base) % m;
        exp /= 2;
    }
    return res;
}

ll modinv(ll n)
{
    ll m = MOD;
    return binexp(n, m - 2);
}

int main() {
    ll t;
    cin >> t;

    while (t--)
    {
        ll n;
        cin >> n;

        vector < ll > a(n);
        for (ll i = 0; i < n; i++)
        {
            cin >> a[i];
        }
        
        vector<ll> prefs(n+1, 0);
        prefs[1] = 1;
        ll ans = 0;
        
        for (ll i = 2; i < n+1; i++)
        {
            ll x = (1 * modinv(i)) % MOD;
            prefs[i] = prefs[i-1] + x;
            prefs[i] %= MOD;
        }

        for (ll i = 0; i < n; i++)
        {
            ll x = max(i, n - i - 1);
            ll y = n - 1 - x;
            
            ll s = (i == 0 || i == n - 1) ? 0 : 1;
            s += prefs[x];
            s%=MOD;
            s *= a[i];
            s %= MOD;
            ans += s;
            ans %= MOD;
        }
        ans %= MOD;
        
        cout << ans << endl;
    }
}
