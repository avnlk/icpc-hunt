#include <bits/stdc++.h>
using namespace std;

const int MOD = 998244353;
const int MN = 200002;
long long inv[MN],t1[MN];
void precompute(){
    inv[1] = 1;
    for(int i =2;i<MN;i++)
    {
        inv[i] = (MOD - MOD/i)*inv[MOD%i]%MOD;
    }
    for(int i = 1;i<MN;i++)
    {
        t1[i] = (t1[i-1]+ inv[i]%MOD);
    }
}

    cin >> t;
