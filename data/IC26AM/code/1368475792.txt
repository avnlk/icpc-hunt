#include <bits/stdc++.h>

#define ll long long
#define mod 998244353

ll modpow(ll x , ll y ){
    if( x==1 or y == 0 ) return 1;
    ll res = modpow( x , y/2);
    res = (res * res )%mod;
    if( y & 1 ) res = res * x % mod;
    return res;
    
}
ll modinv( ll x){
    return modpow(x , mod-2 );
}
