const int mod=998244353;

int power(int x,int a){
    if(a==0){
        return 1;
    }
    if(a==1){
        return x;
    }
    
    if(a%2 == 0){
        int t = power(x,a/2)%mod;
        return (t*t)%mod;
    }

    return (x*power(x,a-1))%mod;
}

int modinv(int x,int p){
    int t = power(x,p-2)%mod;
    return t;
}
