ull power(ull x,ull y,ull p)
{
    ull res=1;
    x=x%p;
    while(y>0)
    {
        if(y&1)res=(res*x)%p;
        y=y>>1;
        x=(x*x)%p;
    }
    return res;
}
ull modinverse(ull n,ull p)
{
return power(n,p-2,p);}
