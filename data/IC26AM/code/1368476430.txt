#include <bits/stdc++.h>
using namespace std;
using int64 = long long;
const int64 MOD = 998244353;


int main() {
	// your code goes here
	int T; cin >> T;
	while(T--){
	    int n; cin >> n;
	    vector<int64> a(n);
	    for(auto &x : a) cin >> x;
	    if (n ==1){
	        cout << 0 << '\n';
	        continue;
	    }
	    vector<int64> inv(n+1);
	    inv[1] = 1;
	    for(int i = 2; i <= n; ++i){
	        inv[i] = MOD - (MOD/i) * inv[MOD % i] % MOD;
	    }
	    vector<int64> H(n+1, 0);
	    for(int i = 1; i <= n; ++i){
	        H[i] = (H[i-1] + inv[i]) % MOD;
	    }
	    int64 answer = 0;
	    for(int i = 0; i < n; ++i){
	        int64 coefficient = (H[i] + H[n-i-1]) % MOD;
	        answer = (answer + (a[i] % MOD)* coefficient)% MOD;
	    }
	    cout << answer << endl;
}}
