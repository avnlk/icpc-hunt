int power(int a,int b){
    if(b==0 or a==1)return 1;
    int half=power(a,b/2);
    int ans=(half*half)%mod;
    if(b&1)ans=(ans*a)%mod;
    return ans;
}
int inv(int a){
    return power(a,mod-2);
}
