# cook your dish here
MOD = 998244353

t = int(input())
for _ in range(t) :
    n = int(input())
    a = list(map(int , input().split()))
    
    inv = [0] * (n + 1)
    inv[1] = 1
    
    for i in range(2,n+1):
        inv[i] = MOD - (MOD//i) * inv[MOD % i ] % MOD
        
    h = [0] * (n+1)
    p = [0] * (n+1)
    
    for i in range(1, n+1):
        h[i] = (h[i-1]  + inv[i]) % MOD
        p[i] = (p[i - 1] + h[i]) % MOD
        
    ans = 0
    
    for i , x in enumerate(a,1):
        c = i-1
        if i<n :
            c += h[n-1] - h[max(i,2) - 1]
            
        if i > 1:
            c += h[n-1] - h[n - min(i,n-1)]
            
        if 1 < i < n :
            c+= p[n-1] - p[n-i] - p[i-1] - inv[2]
            
        ans = (ans + x *c) % MOD
        
    print(ans)
