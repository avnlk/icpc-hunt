#include <bits/stdc++.h>
using namespace std;
const long long MOD=998244353;
long long pow(long long base, long long exp){
    long long res=1;
    while(exp>0){
        if(exp%2){
            res=(res*base)%MOD;
        }
        base=(base*base)%MOD;
        exp/=2;
    }
    return res;
}
