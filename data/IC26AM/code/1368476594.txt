#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 998244353;


ll power(ll a, ll b){
    ll ans = 0;
    while(b){
        if(b & 1) ans = (ans + a) % MOD;
        a = (a + a) % MOD:
        
        b/=2;
        
    }
    return ans;
}

ll mod_div(ll a, ll b){
    return (a *  power(b, MOD-2)) % MOD;
}
