#include <bits/stdc++.h>
using namespace std;
#define MOD 998244353
#define int long long

using ll = long long;
int bpow(int a, int x){
    int p = 1;
    while(x){
        if(x & 1){
            p *= a;
            p %= MOD;
        }
        a *= a;
        a %= MOD;
        x >>= 1;
    }
    return p;
}


const int G = 101;
void ntt(vector<ll>& a, bool invert){
    int n = static_cast<ll>(a.size());
    for(int i = 1, j = 0; i < n; i++){
        int bit = n >> 1;
        for(; j & bit; bit >>= 1){
            j ^= bit;
        }
        j ^= bit;
        if(i < j){
            swap(a[i], a[j]);
        }
    }
    for(int len = 2; len <= n; len <<= 1){
        int wlen = bpow(G, (MOD - 1) / len);
        if(invert){
            wlen = bpow(wlen, MOD - 2);
        }
        for(int i = 0; i < n; i += len){
            int w = 1;
            for(int j = 0; j < len / 2; j++){
                int u = a[i + j];
                int v = (a[i + j + len / 2] % MOD * w) % MOD;
                a[i + j] = (u + v >= MOD ? u + v - MOD : u + v);
                a[i + j + len / 2] = (u >= v ? u - v : u + MOD - v);
                w = (w * wlen) % MOD;
            }
        }
    }
    if(invert){
        int n_inv = bpow(n, MOD - 2);
        for(int& x: a){
            x = (x % MOD * n_inv) % MOD;
        }
    }
}

vector<int> multiply(vector<int> a, vector<int> b){
    int need = static_cast<int>(a.size() + b.size() - 1);
    int n = 1;
    while(n < need) n <<= 1;
    a.resize(n, 0);
    b.resize(n, 0);
    ntt(a, false);
    ntt(b, false);

    for(int i = 0; i < n; i++){
        a[i] = (a[i] % MOD * (b[i] % MOD)) % MOD;
    }
    ntt(a, true);
    a.resize(need);
    return a;
}

vector<int> b;
vector<int> build_polynomial(int l, int r){
    if(l == r){
        // cout << b[l] << "\n";
        return {b[l], 1};
    }
    int mid = l + (r - l) / 2;
    return multiply(build_polynomial(l, mid), build_polynomial(mid + 1, r));
}
