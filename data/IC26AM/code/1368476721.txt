const int inf = 1e9, MOD = 998244353, MAXN = 2e5 + 5;
int fact[MAXN], ifact[MAXN];

void init_factorial() {
    fact[0] = 1;
    for (int i = 1; i < MAXN; i++) fact[i] = (fact[i-1] * i) % MOD;
    for (int i = 0; i < MAXN; i++) ifact[i] = binexp(fact[i], MOD-2, MOD);
}

int binexp(int a, int b, int m=MOD) {
    int ans = 1;
    while (b > 0) {
        if (b&1) ans = (ans * a) % m;
        a = (a * a) % m; b>>=1ll;
    }
    return ans;
}
