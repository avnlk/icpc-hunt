#include <bits/stdc++.h>
using namespace std;
#define int long long
int MOD = 998244353;

int binpow (int a, int b) {
    int ans = 1;
    while (b > 0) {
        if (b & 1) {
            ans = (long long) ((__int128)ans * a % MOD);
        }
        a = (long long) ((__int128)a * a % MOD); 
        b >>= 1;
    }
    return ans;
}

int modinv(int a) {
    return binpow(a, MOD - 2);
}

void solve() {
    int n;
    cin >> n;
    vector<int> a(n);
    for (int i = 0; i < n; i++) cin >> a[i];
    vector<int> pref(n + 1);
    vector<int> suff(n + 1);
    vector<int> pref2(n + 1);
    vector<int> suff2(n + 1);
    for (int i = 1; i <= n; i++) pref[i] = pref[i - 1] + a[i - 1];
    for (int i = n - 1; i >= 0; i--) suff[i] = suff[i + 1] + a[i];
    for (int i = 1; i <= n; i++) pref2[i] = pref2[i - 1] + (i - 1)*(a[i - 1]);
    for (int i = n - 1; i>= 0; i--) suff[i] = suff[i + 1] + (n - i - 1)*(a[i]);
    int ans {};
    for (int l = 2; l <= n; l++) {
        ans = (ans + (pref[l]*modinv(l))%MOD) % MOD;
        ans = (ans + (suff[n - l]*modinv(l))%MOD)%MOD;
        if (n - l > l) {
            ans = (ans + (2*(pref[n - l] - pref[l])%MOD*modinv(l + 1))%MOD)%MOD;
            ans = (ans + (((((2*pref2[l])%MOD)*modinv(l))%MOD)*modinv(l + 1))%MOD)%MOD;
            ans = (ans + (((((2*suff2[n - l])%MOD)*modinv(l))%MOD)*modinv(l + 1))%MOD)%MOD;
        }
        else {
            ans = (ans + (((((2*pref2[n - l])%MOD)*modinv(l))%MOD)*modinv(l + 1))%MOD)%MOD;
            ans = (ans + (((((2*suff2[n-l+1])%MOD)*modinv(l))%MOD)*modinv(l + 1))%MOD)%MOD;
        }
        
    }
    cout << ans << "\n";
}

int32_t main() {
	int t;
	cin >> t;
	while (t--) {
	    solve();
	}

}
