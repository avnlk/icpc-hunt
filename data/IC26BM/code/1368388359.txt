   int t;
   scanf("%d",&t);
   
   while(t--){
       long long c00,c01,c10,c11;
       scanf("%11d %11d %11d %11d",&c00,&c01,&c10,&c11);
       
       long long zero=c00+c10;
       if(c00+c01>0){

       for(long long i=zero;i<len;i++){

       }

}
