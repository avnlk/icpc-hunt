#include <bits/stdc++.h>
using namespace std;
#define ll long long int 

void solve() {
    ll c00, c01, c10, c11;
    if(!(cin >> c00 >> c01 >> c10 >> c11)) return;
    
    string answer = "";
    if(c01 == 0 && c10 == 0) {
        if(c00 > 0) {
            answer += "0";
            for(int i = 0; i < c00; i++) {
                answer += "0";
            }
        } else if(c11 > 0) {
            answer+="1";
            for(int i = 0; i < c11; i++) answer += "1";
        }
        cout << answer << endl;
        return;
    }
    
    answer += "0";
    for(int i = 0; i < c00; i++) answer+= "0";
    
    if(c01 >= c10) {
        ll p = c10;
        for(int i = 0; i < p; i++) {
            answer += "1";
            if(i == 0) {
                for(int j = 0; j < c11; j++) answer += "1";
            }
            answer += "0";
        }
        
        ll rc01 = c01 - c10;
        for(int i=0 ; i< rc01;i++){
            answer += "1";
            if(p == 0&& i== 0){
                for(int j= 0; j<c11 ; j++) answer +="1";
            }
        }
    } else {
        ll p = c01;
        for(int i = 0; i < p; i++) {
            answer += "1";
            if(i == 0) {
                for(int j = 0; j < c11; j++) {
                    answer+= "1";
                }
            }
            answer += "0";
        }
        ll rc10 = c10 - c01;
        for(int i = 0; i < rc10; i++) {
            answer+= "0";
        }
    }
    cout << answer << endl;
}

int main() {
	// your code goes here
	int t; cin >> t;
	while(t--) solve();
	
	return 0;

}
