import sys
def solve(a, b, c, d):
    if b+c == 0:
        if a>0 or d == 0:
            return "0"*(a+1)
        return "1"*(d+1)
    n = b + c + 1
    s = ("01" * n if b >= c else "10"*n)[:n]
    i = s.index("0")
    s = s[:i]+ "0"*a + s[i:]
    j = s.rindex("1")
    s = s[:j]+"1"*d+s[j:]
    return s
data = sys.stdin.read().split()
t = int(data[0])
out = []
for k in range(t):
    a,b,c,d = map(int, data[1+4*k : 5+4*k])
    out.append(solve(a,b,c,d))
print("\n".join(out))
