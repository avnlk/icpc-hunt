#include <stdio.h>
#include<stdlib.h>
#include<string.h>


int main() {
	// your code goes here
	int t;
	scanf("%d",&t);
	while(t--){
	    int c[4];
	    scanf("%d %d %d %d",&c[0],&c[1],&c[2],&c[3]);
	    int n=c[0]+c[1]+c[2]+c[3];
	    char *best=NULL;
	    char *cur=(char *)malloc((n+2)*sizeof(char));
	    for(int last=0;last<4;last++){
	        if(c[last]==0){
	            continue;
	        }
	        int a[4];
	        for(int i=0;i<4;i++)
	        {
	            a[i]=c[i];
	            
	        }
	        a[last]--;
	        int full100=0;
	        int full101=0;
	        int x=(a[3]<a[0])?a[3]:a[0];
	        x=(a[1]<a[0])?a[1]:a[0];
	        full100+=x;
	        a[1]-=x;
	        a[0]-=x;
	        x=(a[3]<a[1])?a[3]:a[1];
	        full101+=x;
	        a[3]-=x;
	        a[1]-=x;
	        x=a[1]/2;
	        full101+=x;
	        a[1]-=2*x;
	        int pos=0;
	        int zeros=a[0]+a[2];
	        for(int i=0;i<zeros;i++){
	            cur[pos++]='0';
	        }
	        for(int i=0;i<full100;i++){
	            cur[pos++]='0';
	            cur[pos++]='0';
	        }
	        for(int i=0;i<full101;i++){
	            cur[pos++]='0';
	            cur[pos++]='1';
	            
	        }
	        int ones=a[1]+a[3];
	        for(int i=0;i<ones;i++){
	            cur[pos++]='1';
	        }
	        if(last==0){
	            cur[pos++]='0';
	            cur[pos++]='0';
	        }
	        else if(last==1){
	            cur[pos++]='0';
	            cur[pos++]='1';
	        }
	        else if(last==2){
	            cur[pos++]='1';
	            cur[pos++]='0';
	        }
	        else{
	            cur[pos++]='1';
	            cur[pos++]='1';
	        }
	        cur[pos]='\0';
	        if(best==NULL||strcmp(cur,best)<0){
	            free(best);
	            best=(char *)malloc((pos+1)*sizeof(char));
	            strcpy(best,cur);
	        }
	        
	            
	    }
	    printf("%s\n",best);
	    free(best);
	    free(cur);
	}
	    return 0;
	

}
