#include <bits/stdc++.h>
using namespace std;

int main() {

    int t;
    cin >> t;
    while(t--)
    {
        int n = 4;
        vector<int> a(n);
        for(int i=0;i<n;i++)
        {
            cin >> a[i];
        }
        int cnt = 0;
        int idx = 3;
        string ans;
        int total = 0;
        for(int i=0;i<n;i++) total += a[i];
        
        while(cnt < total-1)
        {
            if(a[3] > 0 && cnt + 2 <= total-1)
            {
                a[3]--;
                cnt += 2;
            }
            else if(a[2] > 0 && cnt + 1 <= total-1)
            {
                a[2]--;
                ans += "0";
                cnt++;
            }
            else if(a[1] > 0 && cnt + 2 <= total-1)
            {
                a[1]--;
                cnt+=2;
            }
            else
            {
                break;
            }
        }
        vector<string> s = {"00","01","10","11"};
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<a[i];j++)
            {
                ans += s[i];
            }
        }
        reverse(ans.begin(),ans.end());
        while(cnt < total-1)
        {
            ans.pop_back();
            cnt++;
        }
        reverse(ans.begin(),ans.end());
        
        cout << ans << endl;
    }
    
}
