#include <bits/stdc++.h>
using namespace std;
void solve(){
    long long c00,c01,c10,c11;
    cin>>c00>>c01>>c10>>c11;
    string ans="";
    if(c00>0 || c01>0){
        for(int i=0;i<c00;i++) ans+='0';
        for(int i=0;i<c01;i++) ans+="01";
        for(int i=0;i<c11;i++) ans+='1';
        for(int i=0;i<c10;i++) ans+="10";
    } else {
        for(int i=0;i<c10;i++) ans+="10";
        for(int i=0;i<c11;i++) ans+='1';
    }
    if(ans.empty()) ans="00";
    cout<<ans<<"\n";
}
int main() {
	// your code goes here
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin>>t;
    while(t--){
        solve();
    }
    return 0;
}
