import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int t = Integer.parseInt(br.readLine().trim());
		while (t-->0){
		    StringTokenizer st = new StringTokenizer(br.readLine());
		    int c00 = Integer.parseInt(st.nextToken());
		    int c01 = Integer.parseInt(st.nextToken());
		    int c10 = Integer.parseInt(st.nextToken());
		    int c11 = Integer.parseInt(st.nextToken());
		    if (c00 == 0 && c01 ==  0 && c10 ==0){
		        System.out.println("1".repeat(c11 + 1));
		        continue;
		    }
		    StringBuilder sb = new StringBuilder("0");
		    sb.append("0".repeat(c00));
		    
		    if(c01>0){
		        sb.append("1").append("1".repeat(c11));
		        c01--;
		        while (c10 > 0 || c01 > 0){
		            if (c10-->0) sb.append("0");
		            if (c01-->0) sb.append("1");
		        }
		    }else{
		        sb.append("0".repeat(c10)).append("1".repeat(c11));
		    }
		    System.out.println(sb);
		}

	}
}
