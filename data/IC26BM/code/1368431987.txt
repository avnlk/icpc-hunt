import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc=new Scanner(System.in);
		if(!sc.hasNextInt()){
		    return;
		}
		int t=sc.nextInt();
		while(t-->0){
		    int c00=sc.nextInt();
		    int c01=sc.nextInt();
		    int c10=sc.nextInt();
		    int c11=sc.nextInt();
		    int n=c00+c10+c01+c11;
		    int k=n-1;
		    int d=k/2;
		    int rem=k%2;
		    String best=null;
		    if(rem==0){
		        best=getGreedyString(c00,c01,c10,c11,d," ",false);
		        
		    }
		    else{
		        if(c00>0) best=minStr(best,getGreedyString(c00-1,c01,c10,c11,d,"0",true));
		        if(c01>0) best=minStr(best,getGreedyString(c00,c01-1,c10,c11,d,"1",true));
		        if(c10>0) best=minStr(best,getGreedyString(c00,c01,c10-1,c11,d,"0",true));
		        if(c11>0) best=minStr(best,getGreedyString(c00,c01,c10,c11-1,d,"1",true));
		    }
		    System.out.println(best);
		    
		}

	}
	private static String getGreedyString(int c00,int c01, int c10,int c11, int d, String prefix, boolean hasSplit){
	    int take;
	    d-=(take=Math.min(d,c11)); 
	    c11-=take;
	    d-=(take=Math.min(d,c10));
	    c10-=take;
	    d-=(take=Math.min(d,c01));
	    c01-=take;
	    d-=(take=Math.min(d,c00));
	    c00-=take;
	    if(d>0)
	     return null;
	     StringBuilder sb=new StringBuilder(prefix);
	     for(int i=0;i<c00;i++)  sb.append("00");
	     for(int i=0;i<c01;i++)  sb.append("01");
	     for(int i=0;i<c10;i++)  sb.append("10");
	     for(int i=0;i<c11;i++)  sb.append("11");
	     return sb.toString();
}
private static String minStr(String a, String b ){
    if(b==null) return a;
    if(a==null) return b;
    return a.compareTo(b)<0?a:b;

	}
}
