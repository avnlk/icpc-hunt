#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int main() {
	// your code goes here
	int t;
	cin>>t;
	const string tail[] = {"00","01","10","11"};
	while(t--){
	    array<ll,4> cnt;
	    for(auto &x:cnt) cin>>x;
	    ll n = accumulate(cnt.begin(), cnt.end(), 0LL);
	    string ans;
	    bool found=false;
	    ll best0=-1,best01=-1,best10=-1,best1=0;
	    int last=0;
	    for(ll b = 1;2*b<=n+1;++b){
	        ll sing = n+1-2*b;
	        for(int fin=0;fin<4;++fin) {
	            if(cnt[fin] == 0) continue;
	            array<ll,4> x=cnt;
	            --x[fin];
	            ll zerob = x[0] + x[2];
	            ll oneb = x[1] + x[3];
	            ll low = max(0LL,b-1-oneb);
	            ll high = min(b-1,zerob);
	            if(low>high) continue;
	            ll split = max(low,min(high,x[0]));
	            ll edge = min(high,max(low,zerob-sing));
	            if(split>edge) continue;
	            ll mid = b-1-x[1];
	            ll pick = max(split,min(edge,mid));
	            ll zpair = min(x[0],pick);
	            ll npair = pick-zpair;
	            ll rest = b-1-pick;
	            ll opair = min(x[1],rest);
	            ll ypair = rest - opair;
	            ll lonez = min(sing,zerob-pick);
	            ll leadz = 2*zpair+lonez;
	            ll trailone = 2*ypair+sing-lonez;
	            if(!found || leadz>best0 || (leadz==best0 && opair>best01) || 
	            (leadz==best0 && opair==best01 && npair>best10) || 
	            (leadz==best0 && opair == best01 && npair==best10 && fin<last)) {
	                found=true;
	                best0=leadz;
	                best01=opair;
	                best10=npair;
	                best1 = trailone;
	                last = fin;
	            }
	        }
	    }
	    ans.append(best0,'0');
	    for(ll i=0;i<best01;++i) ans += "01";
	    for(ll i=0;i<best10;++i) ans += "10";
	    ans.append(best1,'1');
	    ans += tail[last];
	    cout<<ans<<endl;
	}
return 0;
}
