#include <bits/stdc++.h>
using namespace std;
#define int long long

void Solve() {
    int c00, c01, c10, c11;
    cin >> c00 >> c01 >> c10 >> c11;
    int to_remove = c00 + c01 + c10 + c11 - 1;
    int c0 = 0;
    while(to_remove >= 2 && c11 > 0) {
        c11--;
        to_remove -= 2;
    }
    while(to_remove >= 1 && c10 > 1) {
        c10--;
        to_remove -= 2;
        if(to_remove == -1) {
            to_remove += 1;
            c0 += 1;
        }
    }
    while(to_remove >= 2 && c01 > 0) {
        c01--;
        to_remove -= 2;
    }
    if(to_remove == 1 && c10 >= 1) {
        c10--;
        c0 += 1;
        to_remove--;
    }
    
    while(to_remove >= 1 && c00 > 0) {
        to_remove -= 2;
        c00--;
        if(to_remove == -1) {
            to_remove++;
            c0++;
        }
    }
    
    assert(to_remove <= 1);
    string ans = "";
    if(to_remove == 1) {
        assert(c00 == 0 && c10 == 0 && c0 == 0);
        ans += "1";
        if(c11 > 0) {
            c11--;
        }
        else {
            c01--;
        }
        for(int i = 0; i < c01; i++) {
            ans += "01";
        }
        for(int i = 0; i < c11; i++) {
            ans += "11";
        }
    }
    else {
        for(int i = 0; i < c0; i++) {
            ans += "0";
        }
        for(int i = 0; i < c00; i++) {
            ans += "00";
        }
        for(int i = 0; i < c01; i++) {
            ans += "01";
        }
        for(int i = 0; i < c10; i++) {
            ans += "10";
        }
        for(int i = 0; i < c11; i++) {
            ans += "11";
        }
    }
    cout << ans << "\n";
}

int32_t main() {
	// your code goes here
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    int t;
    cin >> t;
    while(t--) {
        Solve();
    }
    return 0;
}
