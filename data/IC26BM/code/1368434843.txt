import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	static String build(int[] a, int l){
	    int [] c = a.clone();
	    c[l]--;
	   // int foo = Math.min(c[0], c[1]+c[3]);
	   // c[0]-=foo;
	   // int rem = foo;
	   // int x = Math.min(rem, c[3]);
	   // c[3]-=x;
	   // rem -= x;
	   // c[1]-=rem;
	   // int fo1 = Math.min(c[1], (c[1]+c[3])/2);
	   // c[1]-=fo1;
	   // rem = fo1;
	   // x = Math.min(rem, c[3]);
	   // c[3]-=x;
	   // rem -= x;
	   // c[1] -= rem;
	   // StringBuilder sb = new StringBuilder();
	   // int z = c[0]+c[2]+2*foo;
	   int [] f = new int [4];
	   int [] d = new int [4];
	   int x = Math.min(c[0], c[1]+c[3]);
	   f[0] = x;
	   d[3] = Math.min(c[3], x);
	   d[1] = x - d[3];
	   int r01 = c[1] - d[1];
	   int r11 = c[3] - d[3];
	   int y = Math.min(r11, (r01+r11)/2);
	   f[1] = y;
	   int t11 = Math.min(r11, y);
	   d[3] += t11;
	   d[1] += y - t11;
	   int n00 = c[0] - f[0] - d[0];
	   int n01 = c[1] - f[1] - d[1];
	   int n10 = c[2] - f[2] - d[2];
	   int n11 = c[3] - f[3] - d[3];
	   StringBuilder sb = new StringBuilder();
	   int z = n00+n10+2*f[0];
	   
	   
	    while(z-->0){
	        sb.append('0');
	    }
	    for(int i =0;i<f[1];i++){
	        sb.append("01");
	    }
	    int one = n01+n11;
	    while(one-->0){
	        sb.append('1');
	    }
	    if(l == 0) sb.append("00");
	    else if(l == 1) sb.append("01");
	    else if(l ==2) sb.append("10");
	    else sb.append("11");
	    
	    return sb.toString();
	    
	}
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		int t = sc.nextInt();
		while(t-->0){
		    int [] c = new int [4];
		    for(int i = 0;i<4;i++){
		        c[i] = sc.nextInt();
		    }
		    String b = null;
		    for(int i = 0;i<4;i++){
		        if(c[i] == 0) continue;
		        String curr = build(c, i);
		        if(b == null || curr.compareTo(b)<0) b = curr;
		    }
		    System.out.println(b);
		}

	}
}
