t = int(input())
for _ in range(t):
    c = list(map(int,input().split()))
    s = ["00","01","10","11"]
    n = sum(c)
    if n%2:
        k = (n+1)//2
        ans = ""
        for i in range(4):
            x = min(c[i],k)
            ans += s[i]*x
            k -= x 
    else:
        k = n//2
        ans = None
        for j in range(4):
            if c[j]:
                d = c[:]
                d[j] -= 1
                r = s[j][1]
                x = k 
                for i in range(4):
                    z = min(d[i],x)
                    r += s[i]*z
                    x -= z 
                if not x and (ans is None or r < ans):
                    ans = r 
    print(ans)
