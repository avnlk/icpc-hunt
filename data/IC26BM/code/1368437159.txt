#include <bits/stdc++.h>
#define int long long
using namespace std;

signed main() {
    int t;
    cin >> t;
    while (t--) {
        int c00, c01, c10, c11;
        cin >> c00 >> c01 >> c10 >> c11;
        string s;
        if (c00 != 0) {
            c00 -= 1;
            s += "00";
        }
        else if (c01 != 0) {
            c01 -= 1;
            s += "01";
        }
        else if (c10 != 0) {
            c10 -= 1;
            s += "10";
        }
        else if (c11 != 0) {
            c11 -= 1;
            s += "11";
        }
        bool flag = false;
        while (c00 + c01 + c10 + c11 != 0) {
            if (c00 + c01 + c10 + c11 > 1) {
                if (c11 != 0) {
                    c11 -= 1;
                }
                else if (c10 != 0) {
                    c10 -= 1;
                }
                else if (c01 != 0) {
                    c01 -= 1;
                }
                else if (c00 != 0) {
                    c00 -= 1;
                }
            }
            else {
                if (c00 || c10) {
                    cout << "0" << s << endl;
                    flag = true;
                    break;
                }
                else {
                    s = s.substr(1);
                }
            }
            if (c00 != 0) {
                c00 -= 1;
                s.push_back('0');
                s.push_back('0');
            }
            else if (c01 != 0) {
                c01 -= 1;
                s.push_back('0');
                s.push_back('1');
            }
            else if (c10 != 0) {
                c10 -= 1;
                s.push_back('1');
                s.push_back('0');
            }
            else if (c11 != 0) {
                c11 -= 1;
                s.push_back('1');
                s.push_back('1');
            }
        }
        if (flag)
            continue;
        cout << s << endl;
    }
}
