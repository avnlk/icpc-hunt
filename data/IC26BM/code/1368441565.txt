#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
    int t;
    cin>>t;
    while(t--){
        int c00,c01,c10,c11;
        cin>>c00>>c01>>c10>>c11;
        
        int n= c00+c01+c10+c11;
        
        string ans="";
        int pairs= (n-1)/2;
        for(int i=0;i<pairs;i++){
            if(c11 > 0){
                c11--;
                ans= ans+ "11";
            }
            else if(c10>1 && n%2==1){
                c10--;
                ans= ans + "10";
            }else if(n%2==1 && c10>0){
                c10--;
                ans= ans+ "10";
            }else if(c01 > 0){
                c01--;
                ans= ans+ "01";
            }else if(c00 > 0){
                c00--;
                ans= ans+ "00";
            }
        }
        
        if(n%2==0 && c10 >=1){
            c10--;
            ans= ans+"10";
        }
        
        while(c00 > 0){
            c00--;
            ans= ans+ "00";
        }
        while(c01 > 0){
            c01--;
            ans= ans+ "01";
        }
        while(c10>0){
            c10--;
            ans= ans+ "10";
        }
        while(c11 > 0){
            c11--;
            ans= ans + "11";
        }
        
        for(int i=(n-1);i<2*n;i++) cout<<ans[i]<<"";
        cout<<endl;
    }
}
