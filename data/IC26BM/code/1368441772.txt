#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    
    int T;
    if (cin >> T) {
        while (T--){
            vector<long long> v(4);
            cin >> v[0] >> v[1] >> v[2] >> v[3];
            
            sort(v.begin(), v.end());
            
            cout << (v[3] * v[2] + v[1] * v[0]) << "\n";
            
        }
    }
    return 0;
}
