import sys 
input=sys.stdin.readline
STR=["00","01","10","11"]
def solve(c):
    n=sum(c)
    bk=None
    ba=None
    for k in range((n-1)//2+1):
        sn=n-2*k-1
        for last in range(4):
            if c[last]==0:
                continue
            rem=c[:]
            rem[last]-=1
            full=[0]*4
            need=k
            for typ in [0,1,2,3]:
                take=min(rem[typ],need)
                full[typ]=take
                rem[typ]-=take
                need-=take
            if need!=0:
                continue
            single=[0]*4
            need=sn 
            for typ in [0,2,1,3]:
                take=min(rem[typ],need)
                single[typ]=take
                rem[typ]-=take
                need-=take
            if need!=0:
                continue 
            lz=(single[0]+single[2]+2*full[0])
            c01=full[1]
            c10=full[2]
            key=(lz,c01,c10,-last)
            if bk is None or key>bk:
                bk=key
                ans=[]
                ans.append("0"*(single[0]+single[2]+2*full[0]))
                ans.append("01"*full[1])
                ans.append("10"*full[2])
                ans.append("1"*(single[1]+single[3]))
                ans.append("11"*full[3])
                ans.append(STR[last])
                ba="".join(ans)
    return ba
def main():
    t=int(input())
    for _ in range(t):
        ct00,ct01,ct10,ct11=map(int,input().split())
        c=[ct00,ct01,ct10,ct11]
        print(solve(c))
if __name__=="__main__":
    main()
