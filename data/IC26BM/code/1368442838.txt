void solve(){
    cin >> c1 >> c2 >> c3 >> c4; 
    int n = c1+c2+c3+c4;
    int s = c1 + c2 + c4;
    
    if(s == 0){
        cout << string(c3 -1, '0') + "10";
        cout << "\n";
        return;
    }
    
    int k = min(c4 + c2, (s-1)/2);
    int sav1 = min(c1, k+1);
    
    int rem2 = c2 - max(0LL, k - c4);
    
    int sav2 = min(rem2, (k+1) - sav1);
    
    int lz, to;
    
    lz = c1 + c3 + sav1;
    to = (n+1) - lz - 2* sav2;
    
    string res = string(lz, '0');
    for(int i = 0; i < sav2; i++ ) {
        res += "01";
    }
    
    res += string(to, '1');
    cout << res << "\n";
}

    int t; 
    t = 1;
    cin >> t; 
    while(t--){
        solve();
    }
