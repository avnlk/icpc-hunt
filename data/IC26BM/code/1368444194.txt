string func(string& x, string& y){
    int i = 0;
    while(i<x.size() && x[i]==y[i])
        i++;
        
    if(i == x.size())
        return x;
    else{
        int a = x[i] - '0';
        int b = y[i] - '0';
        if(a < b){
            return x;
        }else{
            return y;
        }
    }
    
    return x;
}
