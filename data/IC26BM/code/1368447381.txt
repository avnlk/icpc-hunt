# cook your dish here
t=int(input())
for _ in range(t):
    parts=input().split()
    c00=int(parts[0])
    c01=int(parts[1])
    c10=int(parts[2])
    c11=int(parts[3])
    n=c00+c01+c10+c11
    if c01==0 and c10 ==0:
        if c00>0:
            print("0" *(c00+1))
        else:
            print("1" *(c11+1))
            
        continue
    if c01==0:
        res="0"*(c00+1)+"1"*(c11+1)
        print(res)
        continue
    if c01==0:
        res="0" *c00+"01"*c01
        if c00>0:
            res="0"+res 
        if len(res)>n+1:
            res=res[:n+1]
        print(res)
        continue
    total=2*c00+c01+c10
    numzer=min(n,total)
    numone=(n+1)-numzer
    res="0"*numzer+"1"*numone
    print(res)
