#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int T;
    cin >> T;
    const string type[4]={"00","01","10","11"};
    
    while(T--){
        int c[4];
        cin>>c[0]>>c[1]>>c[2]>>c[3];
        
        string ans;
        bool first= true;
        for(int last =0; last< 4; ++last){
            if (c[last]==0) continue;
            
            int a[4]= {c[0],c[1],c[2],c[3]};
            --a[last];
            
            int k = min({
                a[0]+ a[1],
                a[1]+ a[3],
                (a[1]+ a[1]+a[3])/2
            });
            
            int p00 = min(a[0],k);
            int p01= k- p00;
            
            int zeros =a[0]+a[2]-p00;
            int ones = a[1]+ a[3]-p01 -k;
            
            string cur;
            cur.append(zeros,'0');
            
            for(int i=0; i<p00; ++i)
            cur+="00";
            
            for (int i =0;i<p01; ++i)
            cur+="01";
            
            cur.append(ones,'1');
            
            cur+=type[last];
            if (first|| cur<ans){
                ans= move(cur);
                first=false;
            }
        }
        cout<<ans<<'\n';
    }
    return 0;

}
