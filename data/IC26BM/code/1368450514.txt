import sys

TP = ["00", "01", "10", "11"]

def solve(c):
    if sum(c) == 1:
        return TP[c.index(1)]
    
    best = None
    
    for last in range(4):
        if c[last] == 0:
            continue
        
        d = c[:]
        
        d[last] -=1
        A, B, C, E = d
        ps = max(0, min(B, (B+E-A)//2)) if B+E-A>=0 else 0
        
        for p in (ps, ps+1):
            if p>B:
                continue
            fr = B-p+E-p
            
            if fr<0:
                continue
            
            a2 = min(A, fr)
            
            s = "0" * (A+a2+C) + "01"*p+"1"*(fr-a2)+TP[last]
            if best is None or s<best:
                best=s
            
    return best

def main():
    data = sys.stdin.buffer.read().split()
    t=int(data[0])
    out = []
    for q in range(t):
        out.append(solve([int(v) for v in data[1+4*q: 5+4*q]]))
    
    sys.stdout.write("\n".join(out)+"\n")

main()
