#include <bits/stdc++.h>
using namespace std;

string helper(int x){
    if(x == 3) return "00";
    if(x == 2) return "01";
    if(x == 1) return "10";
    return "11";
}

int main() {
    int t;
    cin>>t;
    while(t--){
        int a,b,c,d;
        cin>>a>>b>>c>>d;
        int op = a + b + c + d - 1;
        int rem; bool odd = false;
        if(op % 2){
            rem = (op-1)/2;
            odd = true;
        }
        else rem = op/2;
        bool extra = false;
        string ans = "";
        vector<int> num = {d,c,b,a};
        for(int i = 0;i<4;i++){
            if(num[i] >= rem){
                num[i]-=rem;
                rem = 0;
                if(i == 1 && num[i] && odd){
                    odd = false;
                    extra = true;
                    num[i]--;
                }
            }
            else{
                rem-=num[i];
                num[i] = 0;
            }
        }
    
        for(int i = 3;i>=0;i--){
            for(int j = 1;j<=num[i];j++){
                ans += helper(i);
            }
        }
        
        if(odd){
            ans.erase(ans.begin());
        }
        if(extra) ans = "0" + ans;
        cout<<ans<<"\n";
    }
}
