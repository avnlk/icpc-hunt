#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    int t;
    cin >> t;
    while(t--) {
        long long c[4];
        cin >> c[0] >> c[1] >> c[2] >> c[3];
        string ans = "";
        for (int i = 0; i < 4; i++) {
            if(c[i] == 0) continue;
            long long h[4] = {c[0], c[1],c[2],c[3]};
            long long f[4] = {0,0,0,0};
            h[i]--;
            f[i]++;
            long long p = min(h[1], h[3]);
            f[1] += p;
            h[1] -= 2 * p;
            p = min(h[0], h[3]);
            f[0] += p;
            h[0] -= p;
            h[3] -= 3;
            p = min(h[0], h[1]);
            f[0] += p;
            h[0] -= p;
            h[1] -= p;
            p = min(h[0], h[2]);
            f[0] += p;
            h[0] -=p;
            h[2] -=p;
            string last = "";
            if(f[3] > 0) {
                last = "11";
                f[3]--;
            } else if(f[2] > 0) {
                last = "10";
                f[2]--;
            } else if(f[1] > 0) {
                last = "01";
                f[1] --;
            }else if(f[0] > 0) {
                last = "00";
                f[0]--;
            }
            string curr_ans = "";
            long long zeroes = max(h[0] + h[2] + 2 * f[0], 0LL) ;
            curr_ans.append(zeroes, '0');
            for(long long i = 0; i < f[1]; i++) {
                curr_ans += "01";
            }
            for(long long i = 0; i < f[2]; i++) {
                curr_ans += "10";
            }
            long long ones  = max(h[1] + h[3] + 2 * f[3], 0LL);
            curr_ans.append(ones, '1');
            curr_ans += last;
            
            if(ans == "" || curr_ans < ans) {
                ans = curr_ans;
            }
        }
        cout << ans << "\n";
    }
    return 0;
}
