#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
	vector<string> s={"00","01","10","11"};
	
	int t;
	cin>>t;
	
	while(t--){
	    vector<long long>c(4);
	    
	    for(auto &x:c)cin>>x;
	    string ans;
	    bool first =1;
	    
	    for(int q=0;q<4;q++)if(c[q]){
	        auto v=c;
	        v[q]--;
	        
	        long long A=v[0],B=v[1],C=v[2],D=v[3];
	        long long k=min({A+B,B+D,(A+B+D)/2});
	        long long x=min(A,k);
	        
	        string curr;
	        curr.reserve(A+B+C+D+2);
	        curr.append(A+C+x,'0');
	        for(int i=0;i<k-x;i++)curr+="01";
	        curr.append(B+D-2*k+x,'1');
	        curr+=s[q];
	        if(first || curr<ans)ans=curr, first=0;
	    }
	    cout<<ans<<endl;
	}

}
