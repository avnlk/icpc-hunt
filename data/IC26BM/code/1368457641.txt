import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
    static String str[] =  {"00","01","10","11"};
    
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc = new Scanner(System.in);
		int t =  sc.nextInt();
		
		while(t-->0){
		    int arr[] = new int[4];
		    
		    for(int i = 0;i<4;i++){
		        arr[i] = sc.nextInt();
		    }
		    
		    String ans =  null;
		    
		    for(int i = 0;i<4;i++){
		        if(arr[i] == 0){
		            continue;
		        }
		        
		        String cand = func(arr,i);
		        
		        if(ans == null || cand.compareTo(ans)<0){
		            ans = cand;
		        }
		    }
		    System.out.println(ans);
		}

	}
	
	
	public static String func(int arr[],int x){
	    int arr2[] = arr.clone();
	    
	    arr2[x]--;
	    
	    int k = Math.min(Math.min(arr2[0]+arr2[1],arr2[1]+arr2[3]),(arr2[0]+arr2[1]+arr2[3])/2);
	    
	    int a = Math.min(arr2[0],k);
	    int b = k-a;
	    
	    int zeroCtr = arr2[0]+arr2[2]+a;
	    
	    int oneCtr = arr2[1]+arr2[3]-b-k;
	    
	    StringBuilder s = new StringBuilder();
	    
	    for(int i = 0;i<zeroCtr;i++){
	        s.append('0');
	    }
	    
	    
	    
	    for(int i = 0;i<b;i++){
	        s.append("01");
	    }
	    
	    for(int i = 0;i<oneCtr;i++){
	        s.append('1');
	    }
	    
	    s.append(str[x]);
	    
	    return s.toString();
	    
	}
}
