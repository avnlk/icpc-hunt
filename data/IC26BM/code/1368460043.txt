import sys

def solve():
    vals = list(map(int, sys.stdin.read().split()))
    if not vals:
        return
    t = vals[0]
    idx = 1
    out = []
    
    for _ in range(t):
        c00, c01, c10, c11 = vals[idx:idx + 4]
        idx += 4
        
        if c01 == 0 and c10 == 0 :
            if c00: 
                out.append("0"*(c00 + 1))
            else: 
                out.append("1" * (c11 + 1))
            continue
        if c01 > c10:
            out.append("0" * c00 + "01" * c10 + "0" + "1" * (c11 + 1))
        elif 10 > c10:
            out.append("0" * (c00 + 1) + "10" * c01 + "1" * (c11 + 1))
        else: 
            cand1 = "0" * (c00 * 1) +"10" * (c01 - 1) + "1" * (c11 + 1)
            cand2 = "0" * c00 + "01" * c01 + "1" * c11 + "0"
            out.append(min(cand1, cand2))
    sys.stdout.write("\n".join(out)+ "\n")
    
if __name__ == "__main__":
    solve()
