# cook your dish here
import sys
input = sys.stdin.readline
t = int(input())

for _ in range(t):
    c00, c01, c10, c11 = map(int, input().split())
    
    if c00 == c01 == c10 == c11 == 0:
        print("")
        continue
    ans = "0"
    
    if c00:
        ans = "0" * (c00 + 1)
    if c01:
        if c00:
            ans += "1" * c01
        else:
            ans = "01" * c01
    if c10:
        if c01:
            ans = "0" + ans
        else:
            ans = "10" * c10
    if c11:
        ans += "1" * c11
    print(ans)
