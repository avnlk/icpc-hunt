#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
	int t;
	cin>>t;
	while(t--){
	    int c00,c01,c10,c11;
	    cin>>c00>>c01>>c10>>c11;
	    int n=c00+c01+c10+c11;
	    int dlt=n-1;
	    string s="";
	    int idx=n-1;
	    while(dlt>0){
	        if(dlt>1){
	           if(c11>0){
	               s+="11";
	               c11--;
	               dlt-=2;
	           }else if(c10>0){
	               s+="10";
	               c10--;
	               dlt-=2;
	           }else if(c01>0){
	               s+="01";
	               c01--;
	               dlt-=2;
	           }else if(c00>0){
	               s+="00";
	               c00--;
	               dlt-=2;
	           }
	        }else if(dlt==1){
	            if(c10>0){
	                s+="10";
	                c10--;
	                dlt-=2;
	            }else if(c00>0){
	                s+="00";
	                c00--;
	                dlt-=2;
	            }else if(c11>0){
	                s+="11";
	                c11--;
	                dlt-=2;
	            }else if(c01>0){
	                s+="01";
	                c01--;
	                dlt-=2;
	            }
	        }
	    }
	    while(c00>0){
	        s+="00";
	        c00--;
	    }
	    while(c01>0){
	        s+="01";
	        c01--;
	    }
	    while(c10>0){
	        s+="10";
	        c10--;
	    }
	    while(c11>0){
	        s+="11";
	        c11--;
	    }
	    
	    for(int i=idx;i<s.size();i++){
	        cout<<s[i];
	    }
	    cout<<endl;
	}

}
