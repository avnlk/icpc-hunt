#include <bits/stdc++.h>
using namespace std;

void solve()
{
    int a, b, c, d;
    cin>>a>>b>>c>>d;
    int s=a+b+c+d-1;
    while(s>1)
    {
        if(d>0)
        {
            d--;
            s-=2;
        }
        else if(c>0)
        {
            c--;
            s-=2;
        }
        else if(b>0)
        {
            b--;
            s-=2;
        }
        else if(a>0)
        {
            a--;
            s-=2;
        }
    }
    
    string o="";
    if(s!=0)
    {
        if(c>0)
        {
            c--;
            o+="0";
        }
        else if(a>0)
        {
            a--;
            o+="0";
        }
        else if(b>0)
        {
            b--;
            o+="1";
        }
        else if(d>0)
        {
            d--;
            o+="1";
        }
    }
    
    while(a--)
    {
        o+="00";
    }
    while(b--)
    {
        o+="01";
    }
    while(c--)
    {
        o+="10";
    }
    while(d--)
    {
        o+="11";
    }
    
    cout<<o<<endl;
}

int main() {
	int t;
	cin>>t;
	while(t--)
	{
	    solve();
	}

}
