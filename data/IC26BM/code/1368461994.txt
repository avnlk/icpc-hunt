#include <bits/stdc++.h>
using namespace std;

#define int int64_t

void testcase() {
    int c00, c01, c10, c11;
    cin >> c00 >> c01 >> c10 >> c11;
    
    vector<int> comb;
    
    int n = c01 + c00 + c11;
    
    for(int i = 0; i < (n-1)/2; i++) {
        if (c11 > 0) {
            comb.push_back(3);
            c11--;
        }
        else if (c01 > 0) {
            comb.push_back(1);
            c01--;
        }
        else {
            comb.push_back(0);
            c00--;
        }
    }
    
    while(c00--) comb.push_back(0);
    while(c01--) comb.push_back(1);
    while(c11--) comb.push_back(3);
    
    string ans = "";
    string tem = "";
    for(int i = 0; i < comb.size(); i++) {
        if (comb[i] == 0) {
            tem += '0';
            tem += '0';
        }
        else if (comb[i] == 1) {
            tem += '0';
            tem += '1';
        }
        else if (comb[i] == 3) {
            tem += '1';
            tem += '1';
        }
    }
    while(c10--) ans += '0';
    
    for(int i = n-1; i < tem.size(); i++) ans += tem[i];
    cout << ans << endl;
}

int32_t main() {
    int t; cin >> t;
    while(t--) testcase();
}
