#include <bits/stdc++.h>
using namespace std;

void solve(){
    int a, b, c, d; cin >> a >> b >> c >> d;
    int n = a+b+c+d;
    string ans = "";
    int join = n-1;
    while(join > 0){
        if(join >= 2 && d > 0){
            d--;
            ans += "11";
            join -= 2;
        }
        else if(c > 0){
            c--;
            ans += "10";
            join--;
        }
        else if(join == 1){
            break;
        }
        else if(b > 0){
            b--;
            ans += "01";
            join--;
        }
        else break;
    }
    while(a > 0)ans += "00", a--;
    while(b > 0)ans += "01", b--;
    while(c > 0)ans += "10", c--;
    while(d > 0)ans += "11", d--;
    cout << ans.substr(n-1) << endl;
    
}

int main() {
	// your code goes here
    int t; cin >> t;
    while(t--){
        solve();
    }
}
