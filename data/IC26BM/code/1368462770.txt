#include <bits/stdc++.h>
using namespace std;
using ll = long long;
int main() {
    // your code goes here
    int t;
    cin >> t;
    string types[4] = {"00","01","10","11"};
    while (t--) {
        ll cnt[4];
        for (int i = 0; i < 4; i++) cin >> cnt[i];
        string ans;
        bool pos = false;
        for(int i=0;i<4;i++){
            if(!cnt[i]) continue;
            ll a = cnt[0] , b = cnt[1] , c = cnt[2] , d = cnt[3];
            if(i==0) a--;
            else if(i==1) b--;
            else if(i==2) c--;
            else d--;
            ll x = min(a,b+d);
            ll u = min(x,d);
            d-=u;
            b-=(x-u);
            ll y = min(b,(b+d)/2);
            ll z = b+d-2*y;
            string s(a+c+x , '0');
            for(int j=0;j<y;j++){
                s+="01";
            }
            s+=string(z,'1');
            s+=types[i];
            if(!pos || s<ans){
                ans = s;
                pos = true;
            }
        }
        cout<<ans<<endl;
    }
}
