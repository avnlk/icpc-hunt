#include <bits/stdc++.h>
using namespace std;

void Solve(){
    vector<int> c(4);
    int n=0;
    for(int i=0; i<4; i++){
        cin>>c[i];
        n+=c[i];
    }
    string ans;
    if(c[0]>c[1]+c[3]){
        for(int i=0; i<n+1; i++){
            ans+="0";
        }
    
    }
    else {
        int x=c[1]+c[3]-c[0];
        for(int i=0; i<c[2]+2*c[0]; i++){
            ans+="0";
        }
        x+=(ans.size()==0);
        if(c[1]-1<=x-c[1]){
            string t;
            for(int i=0; i<c[1]; i++){
                t+="01";
            }
            int sz=ans.size()+t.size();
            for(int i=0; i<n-sz; i++){
                t+="1";
            }
            if(sz<n+1&&x%2==0){
                t="1"+t;
            }
            else if(sz<n+1){
                t=t+"1";
            }
            ans+=t;
        }
        else {
            int sz=ans.size();
            for(int i=0; i<n+1-sz; i++){
                if(i%2==0){
                    ans+=char('0'+(x%2==0));
                }
                else {
                    ans+=char('0'+(x%2==1));
                }
            }
        }
    }
    cout<<ans<<'\n';
}

int main() {
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	int t;
	cin>>t;
	while(t--){
	    Solve();
	}
}
