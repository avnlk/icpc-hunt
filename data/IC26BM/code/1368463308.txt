#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
   ios::sync_with_stdio(false);
   cin.tie(nullptr);
   int t;
   cin>>t;
   const string types[4]={"00","01","10","11"};
   while(t--)   {
       long long c[4];
       for(int i=0;i<4;i++)   {
           cin>>c[i];
       }
       string best;
       bool have=false;
       for(int l=0;l<4;l++)   {
           if(c[l]==0) continue;
           long long d[4]={c[0],c[1],c[2],c[3]};
           d[l]--;
           long long A=d[0], B=d[1], D=d[2],C=d[3];
           long long k=min({(A+B+C)/2,A+B,B+C});
           long long y0=min(A,k);
           long long y1=k-y0;
           long long zeroes=D+A+y0;
           long long ones=B+C-k;
           string s;
           s.reserve(A+B+C+D+3);
           s.append(zeroes,'0');
           for(long long i=0;i<y1;i++)   {
               s+="01";
           }
           s.append(ones-y1,'1');
           s+=types[l];
           if(!have || s<best)  {
               best=move(s);
               have=true;
           }
       }
       cout<<best<<endl;
   }
   return 0;
}
