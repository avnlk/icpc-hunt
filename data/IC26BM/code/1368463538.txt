#include <bits/stdc++.h>
using namespace std;

void solve(){
    int c00,c01,c10,c11;
    cin>>c00>>c01>>c10>>c11;
    int x=c00+c01+c10+c11-1;
    int temp=x;
    string s;
    while(x){
        if(x>=2){
            if(c11){
                s.push_back('1');
                s.push_back('1');
                c11--;
            }
            else if(c10){
                s.push_back('1');
                s.push_back('0');
                c10--;
            }
            else if(c01){
                s.push_back('0');
                s.push_back('1');
                c01--;
            }
            else if(c00){
                s.push_back('0');
                s.push_back('0');
                c00--;
            }
            x-=2;
        }
        else if(x==1){
            if(c10){
                s.push_back('1');
                s.push_back('0');
                c10--;
            }
            x--;
        }
    }
    while(c00){
        s.push_back('0');
        s.push_back('0');
        c00--;
    }
    while(c01){
        s.push_back('0');
        s.push_back('1');
        c01--;
    }
    while(c10--){
        s.push_back('1');
        s.push_back('0');
    }
    while(c11--){
        s.push_back('1');
        s.push_back('1');
    }
    s=s.substr(temp);
    cout<<s<<endl;
}

int main() {
	// your code goes here
    int t;
    cin>>t;
    while(t--){
        solve();
    }
    return 0;
}
