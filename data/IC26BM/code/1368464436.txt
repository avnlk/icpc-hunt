#include <bits/stdc++.h>
using namespace std;

int main() {
	int t;cin >> t;
	while(t--){
	    int c00,c01,c10,c11;cin >> c00 >> c01 >> c10 >> c11;
	    int length=(c00+c01+c10+c11)*2;
	    int tempn=-1;
	    int n=length/2;
	    tempn=n;
	    string ans="";
	    //using c11
	    if((length-(2*c11))>=(tempn+1)){
	        length-=2*c11;
	        n=length/2;
	        c11=0;
	        if((length-(2*c10))>=(tempn+1)){
	            length-=2*c10;
	            n=length/2;
	            c10=0;
	            if((length-(2*c01))>=(tempn+1)){
	                c01=0;
	                length-=2*c01;
	                n=length/2;
	                int use=(length-tempn-1)/2;
	                c00-=use;
	                if(length!=(tempn+1)) ans=ans+'0';
	                else ans=ans+"00";
	                c00--;
	            }else{
	                int use=(length-tempn-1)/2;
	                c01-=use;
	                if(c00>0){
	                    if(length!=(tempn+1)) ans=ans+'0';
	                    else ans=ans+"00";
	                    c00--;
	                }else if(c01>0){
	                    if(length!=(tempn+1)) ans=ans+'1';
	                    else ans=ans+"01";
	                    c01--;
	                }
	            }
	        }else{
	            int use=(length-tempn-1)/2;
	            c10-=use;
	            if(c10>0){
	                c10--;
	                if(length!=(tempn+1)) ans=ans+'0';
	                else ans=ans+"10";
	            }else if(c00>0){
	                if(length!=(tempn+1)) ans=ans+'0';
	                else ans=ans+"00";
	                c00--;
	            }else if(c11>0){
	                c11--;
	                if(length!=(tempn+1)) ans=ans+'1';
	                else ans=ans+"11";
	            }else if(c01>0){
	                c01--;
	                if(length!=(tempn+1))ans=ans+'1';
	                else ans=ans+"01";
	            }
	        }
	    }else{
	        int use=(length-tempn-1)/2;
	        c11-=use;
	        if(c10>0){
	            c10--;
	            if(length!=(tempn+1)) ans=ans+'0';
	            else ans=ans+"10";
	        }else if(c00>0){
	            c00--;
	            if(length!=(tempn+1)) ans=ans+'0';
	            else ans=ans+"00";
	        }else if(c11>0){
	            c11--;
	            if(length!=(tempn+1)) ans=ans+'1';
	            else ans=ans+"11";
	        }else if(c01>0){
	            c01--;
	            if(length!=(tempn+1)) ans=ans+'1';
	            else ans=ans+"01";
	        }
	    }
	    //cout<<ans<<" "<<c00<<" "<<c01<<" "<<c10<<" "<<c11<<endl;
	        for(int i=0;i<c00;i++) ans+="00";
	        for(int i=0;i<c01;i++) ans+="01";
	        for(int i=0;i<c10;i++) ans+="10";
	        for(int i=0;i<c11;i++) ans=ans+"11";
	    cout<<ans<<endl;
	}
}
