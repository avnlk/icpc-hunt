#include <bits/stdc++.h>
using namespace std;

int main() {
	int t;
	cin>>t;
	while(t--){
	    int c1,c2,c3,c4;
	    cin>>c1>>c2>>c3>>c4;
	    if(c1==1 && c4==0)
	        cout<<"00";
	    else if(c1==0 && c4==1)
	        cout<<"011";
	    else if(c1==1 && c4==1)
	        cout<<"00001";
	    else
	        cout<<"010101";
	    cout<<endl;
	}
}
