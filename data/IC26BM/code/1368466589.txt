#include <bits/stdc++.h>
using namespace std;

int main() {
	int t;
	cin >> t;
	
	while(t--){
	    int a,b,c,d;
	    cin >> a >> b >> c >> d;
	    
	    if(a==0 && b==0 && d==0){
	        while(c>1){
	            cout << "0";
	            c--;
	        }
	        cout << "10" << '\n';
	    }
	    else{
	        for(int i=0;i<c;i++) cout << "0";
	        
	        int n = a+b+d;
	        int rem = n-1;
	        
	        if(rem%2==0){
	            string s;
	            
	            while(a>0){
	                s += "00";
	                a--;
	            }
	            while(b>0){
	                s += "01";
	                b--;
	            }
	            while(d>0){
	                s += "11";
	                d--;
	            }
	           
	            for(int i=0;i<n+1;i++) cout << s[i];
	            cout << '\n';
	        }
	        else{
	            while(rem>1 && d>0){
	                d--;
	                rem -= 2;
	            }
	            while(rem>1 && b>0){
	                b--;
	                rem -= 2;
	            }
	            while(rem>1 && a>0){
	                a--;
	                rem -= 2;
	            }
	            
	            if(a>0){
	                cout << "0";
	                a--;
	            }
	            else if(d>0){
	                cout << "1";
	                d--;
	            }
	            else{
	                cout << "1";
	                b--;
	            }
	            
	            string s;
	            
	            while(a>0){
	                s += "00";
	                a--;
	            }
	            while(b>0){
	                s += "01";
	                b--;
	            }
	            while(d>0){
	                s += "11";
	                d--;
	            }
	            
	            cout << s << '\n';
	        }
	    }
	}
}
