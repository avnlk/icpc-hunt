#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
	int t;cin>>t;
	while(t--){
	    int a,b,c,d;
	    cin>>a>>b>>c>>d;
	    int n=a+b+c+d;
	    bool ok1=false,ok2=false,ok3=false,ok4=false;
	    int sum=2*d;
	    if(sum>n-1){
	        ok1=true;
	    } sum+=2*c; if(sum>n-1){
	        ok2=true;
	    } sum+=2*b; if(sum>n-1){
	        ok3=true;
	    } string ans="";
	    if(ok1){
	        d=d-(n-1)/2;
	        if(c>0){
	        if((n-1)%2==1){
	            c--;ans+='0';
	        }}
	    } else if(ok2){
	        
	        c=c-(n-1-2*d)/2;d=0;
	        if(c>0){
	        if((n-1)%2==1){
	            c--;ans+="0";
	        }}
	    }
	    else if(ok3){
	         
	        b=b-(n-1-2*d-2*c)/2;c=0;d=0;
	        if((n-1)%2==1){
	            a--;ans+="0";
	        }
	    } else{
	        a=a-(n-1-2*c-2*d-2*b)/2;b=0;c=0;d=0;
	        if((n-1)%2==0){
	            a--;ans+="0";
	        }
	    }
	    for(int i=0;i<a;i++){
	        ans+="00";
	    } for(int i=0;i<b;i++){
	        ans+="01";
	    } for(int i=0;i<c;i++){
	        ans+="10";
	    } for(int i=0;i<d;i++){
	        ans+="11";
	    } cout<<ans<<endl;
	}

}
