#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
	// your code goes here
    ll t;
    cin >> t;
    while(t--){
        ll c1,c2,c3,c4;
        cin >> c1 >> c2 >> c3 >> c4;
        ll res = c1+c2+c3+c4-1;
        while(res > 1 && c4 > 0){
            res -= 2;
            c4--;
        }
        while(res > 1 && c3 > 0){
            res -= 2;
            c3--;
        }
        while(res > 1 && c2 > 0){
            res -= 2;
            c2--;
        }
        while(res > 1 && c1 > 0){
            res -= 2;
            c1--;
        }
        string ans = "";
        if(res == 1){
            if(c3 > 0){
                c3--;
                ans += '0';
            }
            else if(c1 > 0){
                c1--;
                ans += '0';
            }
            else if(c4 > 0){
                c4--;
                ans += '1';
            }
            else {
                c2--;
                ans += '1';
            }
        }
        for(int i=0;i<c1;i++){
            ans += "00";
        }
        for(int i=0;i<c2;i++){
            ans += "01";
        }
        for(int i=0;i<c3;i++){
            ans += "10";
        }
        for(int i=0;i<c4;i++){
            ans += "11";
        }
        cout << ans << endl;
        
    }
}
