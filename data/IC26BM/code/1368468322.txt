# cook your dish here
t=int(input())
for _ in range(t):
    a,b,c,d=map(int,input().split())
    n=a+b+c+d-1
    r=0
    while n>0:
        if d>0:
            if n>1:
                d-=1
                n-=2
            else:
                if c>0:
                    c-=1
                    print('0'+a*'00'+b*'01'+c*'10'+d*'11')
                else:
                    print('0'+'00'*(a-1)+b*'01'+c*'10'+d*'11')
                r=1
                break
        elif c>0:
            if n>1:
                n-=2
                c-=1
            else:
                print('0'+a*'00'+b*'01'+(c-1)*'10')
                r=1
                break
        elif b>0:
            if n>1:
                n-=2
                b-=1
            else:
                print('0'+'00'*(a-1)+b*'01')
                r=1
                break
        else:
            if n>1:
                n-=2
                a-=1
            else:
                print('00'*(a-1)+'0')
                r=1
                break
    if not r:
        print(a*'00'+b*'01'+c*'10'+d*'11')
