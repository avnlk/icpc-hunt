#include <bits/stdc++.h>
using namespace std;

int main() {
	// your code goes here
	int t;
	cin>>t;
	while(t--){
	    int c00, c01, c10, c11;
	    cin>>c00>>c01>>c10>>c11;
	    
	    int n=c00+c01+c10+c11;
	       long long k=0;
	       
	    while(c11 && k+2<=n-1){
	        c11--;
	        k+=2;
	    }
	    
	    while(c01 && k+2<=n-1){
	        c01--;
	        k+=2;
	    }
	    
	    while(c10 && k+2<=n-1){
	        c10--;
	        k+=2;
	    }
	    
	    while(c00 && k+2<=n-1){
	        c00--;
	        k+=2;
	    }
	    string s="";
	    
	    if(k+1==n-1){
	        if(c10){
	            c10--;
	            s+='0';
	        }
	        else if(c00){
	            c00--;
	            s+='0';
	        }
	        else if(c11){
	            c11--;
	            s+='1';
	        }
	        else{
	            c01--;
	            s+='1';
	        }
	    }
	    
	    while(c00){
	        s+="00";
	        c00--;
	    }
	    while(c01){
	        s+="01";
	        c01--;
	    }
	    while(c10){
	        s+="10";
	        c10--;
	        
	    }
	    while(c11){
	        s+="11";
	        c11--;
	    }
	    
	    cout<<s<<"\n";
	}

}
