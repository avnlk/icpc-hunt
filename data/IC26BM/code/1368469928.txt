#include <bits/stdc++.h>
using namespace std;
void solve(){
    vector<int> v(4);
    int sum = 0;
    for(int i = 0;i<4;i++){
        cin>>v[i];
        sum+=v[i];
    }
    // cout<<sum<<endl;
    bool flag = (sum%2)==0;
    sum-=1;
    sum = sum/2;
    // cout<<sum<<endl;
    if(v[3]>=sum){
        v[3]-=sum;
        sum =  0;
        
    }else{
        sum-=v[3];
        v[3] = 0;
        
    }
    if(flag){
        if(v[2]>=sum && sum>0){
            v[2]-=sum-1;
            sum = 1;
        }else{
            sum-=v[2]-1;
            v[2] = 1;
            
        }
        
    }else{
        if(v[2]>=sum ){
            v[2]-=sum;
            sum = 0;
        }else{
            sum-=v[2];
            v[2] = 0;
            
        }
        
    }
    if(v[1]>=sum){
        v[1]-=sum;
        sum = 0;
    }else{
        sum-=v[1];
        v[1] = 0;
        
    }
    if(v[0]>=sum){
        v[0]-=sum;
        sum = 0;
    }else{
        sum-=v[0];
        v[0] = 0;
        
    }
    string ans ="";
    if(flag && v[2]>0){
        
        ans+="10";
        v[2]--;
        
    }
    // cout<<sum<<"&"<<endl;
    // for(int i = 0; i < 4; i++){
    //     std::cout << v[i] << " ";
    // }
    // cout << endl;
    while(v[0]--){
        ans+="00";
    }
    while(v[1]--){
        ans+="01";
    }
    while(v[2]--){
        ans+="10";
    }
    while(v[3]--){
        ans+="11";
    }
    // cout<<flag<<endl;
    if(flag){
    cout<<ans.substr(1)<<endl;
    return;
        
    }
    cout<<ans<<endl;
    
    
    
    
}
int main() {
	int t;
	cin>>t;
	while(t--){
	    solve();
	}

}
