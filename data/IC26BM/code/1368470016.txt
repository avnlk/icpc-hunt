import sys
input=sys.stdin.readline

for _ in range(int(input())):
    c=list(map(int,input().split()))
    n=sum(c)
    
    if n==1:
        print(["00","01","10","11"][c.index(1)])
        continue
    
    a=["00","01","10","11"]
    ans=None
    
    for i in range(4):
        if c[i]:
            d=c[:]; d[i]-=1 
            s=a[i][1]+''.join(a[j]*d[j] for j in range(4))
            if ans is None or s<ans: ans=s 
            
    print(ans)
