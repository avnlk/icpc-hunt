#include <bits/stdc++.h>
using namespace std;

int main() {
    int t;
    cin >> t;
    while(t--) {
        vector<int> c(4);
        for(auto &x: c) cin >> x;
        int n = c[0] + c[1] + c[2] + c[3];
        vector<string> b = {"00", "01", "10", "11"};
        if(n & 1) {
            string ans;
            int rem = (n + 1) / 2;
            for(int i = 0; i < 4; i++) {
                for(int j = 0; j < min(c[i], rem); j++) {
                    ans += b[i];
                }
                rem -= min(c[i], rem);
            }
            cout<<ans<<"\n";
         } else {
             string ans;
             for(int i = 0; i < 4; i++) if(c[i]) {
                 c[i] --;
                 string s;
                 int rem = (n / 2);
                 for(int j = 0; j < 4; j++) {
                     for(int l = 0; l < min(c[j], rem); l++) {
                         s += b[j];
                     }
                     rem -= min(c[j], rem);
                 }
                 string curr = b[i][1] + s;
                 if(ans == "" || curr < ans) ans = curr;
                 c[i]++;
             }
             cout<<ans<<"\n";
         }
    }
}
