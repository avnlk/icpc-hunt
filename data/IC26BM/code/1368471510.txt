#include <bits/stdc++.h>
using namespace std;

void solve() {
    int a, b, c, d;
    cin >> a >> b >> c >> d;
    
    int n=a+b+c+d;
    int zeroes=c;
    int del=(n-1-c)/2;
    
    int rem=(n-1-c)%2;
    
    if (d>del) {
        d-=del;
        if (rem) c--;
        for(int i=0; i<2*a+c; i++) cout << '0';
        for(int i=0; i<b; i++) cout << "01";
        for(int i=0; i<d; i++) cout << "11";
        cout << endl;
    } else {
        rem+=n-1-c-2*d;
        
        if (rem/2>b) b=0, c-=2*b;
        else {
            b-=rem/2;
            c-=rem%2;
        }
        
        for(int i=0; i<2*a+c; i++) cout << '0';
        for(int i=0; i<b; i++) cout << "01";
        cout << endl;
    }
    
}
int main() {
	// your code goes here
    int t;
    cin >> t;
    while(t--) {
        solve();
    }
}
