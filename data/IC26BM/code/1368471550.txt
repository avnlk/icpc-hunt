void solve()
{
    vector<int>arr(4);
    string s="";
    for(int i=0;i<4;i++)
    {
        cin>>arr[i];
    }
    int n=arr[0]+arr[1]+arr[3];
    if(arr[3]+1>=arr[0]+arr[1])
    {
        for(int i=0;i<arr[2];i++)
        s+='0';
        for(int i=0;i<arr[3]+1-arr[0]-arr[1];i++)
        {
            s+='1';
        }
        for(int i=0;i<arr[0];i++)
        {
            s+="00";
        }
        for(int i=0;i<arr[1];i++)
        {
            s+="01";
        }
    }
    else
    {
        
        if(2*arr[1]+2*arr[3]>=n-1)
        {
        for(int i=0;i<arr[2];i++)
        s+='0';
        int c=n-1-(2*arr[3]);
        if(c%2==1)
        s+='0';
        for(int i=0;i<arr[0]-c%2;i++)
        {
            s+="00";
        }
        for(int i=0;i<arr[1]-c/2;i++)
        {
            s+="01";
        }
        }
        else
        {
            int c=n-1-2*arr[3]-2*arr[1];
        for(int i=0;i<arr[2];i++)
        s+='0';
        
        if(c%2==1)
        s+='0';
        for(int i=0;i<arr[0]-c/2-c%2;i++)
        {
            s+="00";
        }
            
        }
        
    }
    cout<<s<<endl;
}
