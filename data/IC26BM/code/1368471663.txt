#include <bits/stdc++.h>
using namespace std;
struct Rep{
    int z,a,b,o,x,last;
};
struct Seg{
    int type,first,len;
};
std::vector<Seg>build(const Rep& s){
    vector<Seg>v;
    auto add=[&](int type,int first,int len){
        if(len)v.push_back({type,first,len});
    };
    add(0,0,s.z);
    add(1,0,2*s.a);
    add(1,1,2*s.b);
    add(0,1,s.o);
    add(0,1,2*s.x);
    
    add(0,s.last/2,1);
    add(0,s.last%2,1);
    vector<Seg>w;
    for(auto q:v){
        if(!w.empty() && q.type==0 && w.back().type==0 && q.first==w.back().first){
            w.back().len+=q.len;
        }else{
            w.push_back(q);
        }
    }
    return w;
}
int getChar(const Seg & s, int p){
    if(s.type ==0){return s.first;}
    return s.first^(p&1);
}
bool lessRep(const Rep & A, const Rep& B){
    auto  a=build(A);
    auto b=build(B);
    
    int i=0,j=0;
    int pa=0,pb=0;
    while(i<(int)a.size() && j<(int) b.size()){
        int ra=a[i].len-pa;
        int rb=b[j].len-pb;
        int m=min(ra,rb);
        int lcp=0;
        
        if(a[i].type==0 && b[j].type==0){
            if(a[i].first==b[j].first)lcp=m;
            }else if(a[i].type==1 && b[j].type==1){
                if(a[i].first==b[j].first)lcp=m;
            }else{
                int ca=getChar(a[i],pa);
                int cb=getChar(b[j],pb);
                if(ca==cb)lcp=min(1,m);
            }
            if(lcp<m){
                return getChar(a[i],pa+lcp)<getChar(b[j],pb+lcp);
            }
            pa+=m;
            pb+=m;
            if(pa==a[i].len){
                ++i;
                pa=0;
            }
            if(pb==b[j].len){
                ++j;
                pb=0;
            }
        
    }
    return false;
}
