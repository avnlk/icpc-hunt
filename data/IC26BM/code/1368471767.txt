#include <bits/stdc++.h>
using namespace std;


void solve(){
    int c00, c01, c10 ,c11;
    cin >> c00 >> c01 >> c10 >> c11;
    int n = c00 + c01 + c10 + c11;
    int rem = n-1;
    string add = "";
    while(rem>0){
        if(rem >1){
            if(c11 > 0){
                c11 -=1;
                rem -=2;
                continue;
            }
            if (c10 >0){
                c10 -= 1;
                rem -=2;
                continue;
            }
            if (c01 >0){
                c01 -= 1;
                rem -=2;
                continue;
            }
            if (c00 >0){
                c00 -= 1;
                rem -=2;
                continue;
            }
        }
        else if(rem == 1){
            if (c10 >0){
                c10 -= 1;
                rem -=1;
                add = "0";
                continue;
            }
            if (c00 >0){
                c00 -= 1;
                rem -=2;
                add = "0";
                continue;
            }
            if(c11 > 0){
                c11 -=1;
                rem -=1;
                add = "1";
                continue;
            }
            if (c01 >0){
                c01 -= 1;
                rem -=1;
                add = "1";
                continue;
            }
            
        }
    }
    for(int i = 0 ; i < c00 ; i ++){
        add += "00";
    }
    for(int i = 0 ; i < c01 ; i ++){
        add += "01";
    }
    for(int i = 0 ; i < c10 ; i ++){
        add += "10";
    }
    for(int i = 0 ; i < c11 ; i ++){
        add += "11";
    }
    cout <<add << "\n";
}
int main() {
    int t;
    cin >> t;
    while(t--){
        solve();
    }

}
