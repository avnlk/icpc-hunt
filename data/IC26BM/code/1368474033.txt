#include <bits/stdc++.h>
using namespace std;
#define ll long long

void solve(){
    ll a,b,c,d;
    cin>>a>>b>>c>>d;
    
    ll n=a+b+c+d;
    ll temp=n-1;
    if (temp%2==0) {
        while (1) {
           if (temp>=2) {
               if (d>0) d--;
               else if (c>0) c--;
               else if (b>0) b--;
               else a--;
               temp-=2;
           }
           else break;
        }
    }
    else {
        while (1) {
           if (temp>=2) {
               if (d>0) d--;
               else if (c>1) c--;
               else if (b>0) b--;
               else a--;
               temp-=2;
           }
           else break;
        }
    }
    bool flag=false;
    if (temp) {
        if (c>0) c--;
        else if (a>0) a--;
        else if (d>0) {
            d--;
            flag=true;
        }
        else {
            b--;
            flag=true;
        }
    }
    
    string ans="";
    if (temp) {
        if (flag) ans+="1";
        else ans+="0";
    }
    
    while (a>0) {
        ans+="00";
        a--;
    }
    while (b>0) {
        ans+="01";
        b--;
    }
    while (c>0) {
        ans+="10";
        c--;
    }
    while (d>0) {
        ans+="11";
        d--;
    }
    
    cout<<ans<<'\n';
    
    
}


int main() {
	// your code goes here
    ll t;
    cin>>t;
    
    while(t--){
        solve();
    }
}
