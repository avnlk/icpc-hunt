t = int(input())
for z in range(t):
    a, b, c, d = map(int, input().split())
    n = a+b+c+d
    lim = n-1
    add0start = False
    add1start = False
    agone = 0
    bgone = 0
    cgone = 0
    dgone = 0
    while lim > 1:
        if d > 0:
            d -= 1
        elif c > 0:
            c -= 1
        elif b > 0:
            b -= 1
        else:
            a -= 1
        if a > 0:
            a -= 1
            agone += 1
        elif b > 0:
            b -= 1
            bgone += 1
        elif c > 0:
            c -= 1
            cgone += 1
        else:
            d -= 1
            dgone += 1
        lim -= 2
    if lim == 1:
        if c >= 1:
            c -= 1
            add0start = True
        elif a >= 1:
            a -= 1
            add0start = True
        elif d >= 1:
            d -= 1
            add1start = True
        elif b >= 1:
            b -= 1
            add1start = True
    res = []
    if add0start == True:
        res = ["0"]
    if add1start == True:
        res = ["1"]
    res.extend(["00"]*(a+agone))
    res.extend(["01"]*(b+bgone))
    res.extend(["10"]*(c+cgone))
    res.extend(["11"]*(d+dgone))
    print("".join(res))
