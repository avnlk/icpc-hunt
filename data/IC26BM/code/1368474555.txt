#include <bits/stdc++.h>
using namespace std;

int main() {
	ios_base::sync_with_stdio(false);
	cin.tie(NULL);
	int t;
	cin>>t;
	while(t--){
	    long long c00,c01,c10,c11;
	    cin>>c00>>c01>>c10>>c11;
	    long long n = c00+c01+c10+c11;
	    if(n==1){
	        if(c00)cout<<"00\n";
	        else if(c01)cout<<"01\n";
	        else if (c10)cout<<"10\n";
	        else cout <<"11\n";
	        continue;
	    }
	    long long total0=2*c00+c01+c10;
	    long long total1=2*c11+c01+c10;
	    long long max0=min(n+1,total0);
	    bool can_zero_star_one_star=(c00+c01>0)||(c01>0 &&n>1);
	    if(can_zero_star_one_star && max0 == n+1){
	        cout<<string (n+1,'0')<<"\n";
	    }
	    else if(can_zero_star_one_star && max0>0){
	        cout<<string(max0,'0')<<string(n+1-max0,'1')<<"\n";
	        
	    }else{
	        string res;
	        res.reserve(n+1);
	        bool start0=(c00+c01>0);
	        char cur = start0?'0':'1';
	        for(long long i=0;i<=n;i++){
	            res+=cur;
	            cur =(cur=='0'?'1':'0');
	        }
	        cout<<res<<"\n";
	    }
	}
	return 0;

}
