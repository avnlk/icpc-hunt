	while(t--){

	     int n=a+b+c+d;
	     int nn=n-1;
	     string s="";

	         d-=val;

	        val=min(b,nn/2);

	        val=min(a,nn/2);

	        nn-=val*2;
	        if(n%2==0){
	            if(c) {
	             c--;
	             s+="0";
	         }
	         else if(a){
	             a--;
	             s+="0";
	         }
	         else if(d){
	             d--;
	             s+="1";
	         }
	         else{
	             b--;

	         }
	        }

	     }

	     }
	     while(d--){

	     }

	}
   
}
