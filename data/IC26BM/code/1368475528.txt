        string s="";
        if(c3>=m/2){

            if(m==1){

                    }
                    else{
                        c1--;m=0;s+='1';
                    }
                }
            }
        }
        else{
            m-=2*c3;c3=0;
            if(m&1){
                ll m1=m+1;
                if(c2>=m1/2){
                    c2-=m1/2;m=0;s+='0';
                }

                    m-=2*c2;c2=0;

                            c0--;m=0;s+='0';

                        }

                        m-=2*c1;c1=0;

                        m-=c1*2;c1=0;

                    }
                }
            }
        }
        while(c0--){
            s+='0';s+='0';
        }
        while(c1--){
            s+='0';s+='1';
        }
        while(c2--){
            s+='1';s+='0';
        }
        while(c3--){
            s+='1';s+='1';
        }
        cout<<s<<endl;
    }
