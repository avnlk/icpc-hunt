#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define vec vector<ll>


string solve( int c00, int c01, int c10, int c11 )
{
    int n = c00 + c01 + c10 + c11;
    if ( n == 1 )
    {
        if ( c00 ) return "00";
        if ( c01 ) return "01";
        if ( c10 ) return "10";
        return "11";
    }
    string ans;
    int z = 2 * c00 + c10;
    if ( z >= n - 1 )
    {
        ans.append( n-1, '0' );
        int rem = z-(n-1);
        if ( rem > 0 )
        {
            ans += '0';
            rem--;
        }
        else 
        {
            ans += '1';
        }
        if ( c00 + c10 > 0 )
        {
            ans += '0';
        }
        else ans += '1';
        return ans;
    }
    ans.append( z, '0' );
    int left = n + 1 - z;
    
    while ( left > 0 )
    {
        if ( c01 > 0 )
        {
            ans += '0';
            left--;
            if ( left > 0 )
            {
                ans += '1';
                left--;
            }
            c01--;
        }
        else 
        {
            ans += '1';
            left--;
        }
    }
    return ans;
}

int main() {
	int t;
	cin >> t;
	while ( t-- )
	{
	    int c1, c2, c3, c4;
	    cin >> c1 >> c2 >> c3 >> c4;
	    cout << solve( c1, c2, c3, c4 ) << '\n';   
	}
    return 0;
}
