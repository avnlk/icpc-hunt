#include <bits/stdc++.h>
#include <string>
using namespace std;

int main() {
	// your code goes here
    int n;
    cin>>n;
    for(int i=0;i<n;i++){
        int c00,c01,c10,c11;
        cin>>c00>>c01>>c10>>c11;
        
        int m = c00+c01+c10+c11;
        int l=m*2,f=-1;
        string s="";
        while(m){
            if(m>2){
                if(c11){
                    s+="11";
                    c11--;
                }
                else if(c00){
                    s+="00";
                    c10--;
                }
                else if(c10){
                    s+="10";
                    c00--;
                }
                else{
                    s+="01";
                    c01--;
                }
            }
            else{
                if(c10){
                    s+="10";
                    c10--;
                }
                else if(c00){
                    s+="00";
                    c00--;
                }
                else if(c01){
                    s+="01";
                    c01--;
                }
                else{
                    s+="11";
                    c11--;
                }
            }
            f++;
            m--;
        }
        for(int i=f;i<l;i++){
            cout<<s[i];
        }
        cout<<endl;
    }
    
}
