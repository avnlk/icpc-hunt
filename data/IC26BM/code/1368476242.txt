import heapq
t = int(input())


def heap_up_A(h, item):
    h.append(item)
    i = len(h)-1
    while i > 0:
        if h[i][1::] < h[(i-1)//2][1::]:
            h[i], h[(i-1)//2] = h[(i-1)//2], h[i]
            i = (i-1)//2
        else:
            break
    return h
    
def heap_up_B(h, item):
    h.append(item)
    i = len(h)-1
    while i > 0:
        if h[i] < h[(i-1)//2]:
            h[i], h[(i-1)//2] = h[(i-1)//2], h[i]
            i = (i-1)//2
        else:
            break
    return h

def extract_min_A(h):
    i = 0
    z = h[0]
    h[0] = h[-1]
    h.pop()
    n = len(h)
    while 2*i + 1 < n:
        if 2*i + 2 < n:
            if h[2*i+1][1::] > h[2*i+2][1::] and h[2*i+2][1::] < h[i][1::]:
                h[i], h[2*i + 2] = h[2*i + 2], h[i]
                i = 2*i + 2
            elif h[2*i+1][1::] < h[i][1::]:
                h[i], h[2*i+1] = h[2*i+1], h[i]
                i = 2*i+1
            else:
                break
        elif h[2*i+1][1::] < h[i][1::]:
                h[i], h[2*i+1] = h[2*i+1], h[i]
                i = 2*i+1
        else:
            break
    return z

                h[i], h[2*i + 2] = h[2*i + 2], h[i]

for _ in range(t):
    a,b,c,d = list(map(int, input().split()))
    lst_A = ['00']*a + ['01']*b + ['10']*c + ['11']*d
    lst_B = ['00']*a + ['01']*b + ['10']*c + ['11']*d
    
    dic = {}

    dic['01'] = b
