import sys
i=iter(map(int, sys.stdin.buffer.read().split()))
r=[];
for _ in range(next(i)):
    a=[next(i) for _ in range(4)]
    n=sum(a)-1
    b=[]
    for j, s in enumerate(("00", "01", "10", "11")):
        if a[j]:
            a[j] -=1
            z=min(n, 2*a[0]+a[2])
            k=min(a[1], (n-z)//2)
            b.append("0" *z+"01" * k+"1"*(n-z-2*k)+s)
            a[j]+=1
            
    r.append(min(b))
    sys.stdout.write("\n".join(r))
