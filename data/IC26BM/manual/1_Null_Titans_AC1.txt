import sys
def s1(c):
    be=None 
    for L,s in enumerate(("00","01","10","11")):
        if c[L]==0:
            continue
        w,x,y,z=c
        w-=L==0; x-=L==1; y-=L==2;z-=L==3
        a=min(w,x+z)
        b=min(x,(x+z-a)//2)
        ans="0"*(w+y+a)+"01"*b+"1"*(x+z-a-2*b)+s
        if be is None or ans< be:
            be = ans
    return be
        
dt=sys.stdin.buffer.read().split()
t=int(dt[0])
print("\n".join(s1(list(map(int,dt[1+4*i:5+4*i]))) for i in range(t)))
