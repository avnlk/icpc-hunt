#include <bits/stdc++.h>
using namespace std;


int main(){
    
ios::sync_with_stdio(false);
cin.tie(nullptr);
int t;
cin>>t;
while(t--){
   int cnt[4];
    int n=0;
    for(int i =0; i<4;i++){
        cin>>cnt[i];
        n+=cnt[i];
        
    }
    string answer;
    for ( int last=0; last<4;++last){
        if (cnt[last]==0) continue;
        --cnt[last];
        int a = cnt[0];
        int b = cnt[1];
        int c = cnt[2];
        int d  = cnt[3];
        int length =n-1;
        auto zeros = [&](int pairs){
            return a+c+min({a,length/2-pairs , b+d -2 *pairs});
            
        };
        auto consider = [&](int pairs){
            int z = zeros(pairs);
            int ones = length -z-2*pairs;
            string current;
            current.reserve(n+1);
            current.append(z,'0');
            for (int i=0;i< pairs; ++i) {
                current+="01";
                
            }
            current.append(ones,'1');
            current+=char('0'+last/2);
            current+=char ('0'+last%2);
            if (answer.empty()|| current<answer) {
                answer=current;
            }
        };
        consider(0);
        int limit = min(b,length/2);
        if (limit>=1) {
            int target = zeros (1);
            int low = 1;
            int high =limit;
            while (low<high){
                int mid = low + (high-low+1)/2 ;
                if (zeros(mid)==target){
                    low = mid;
                }
                else {
                    high=mid-1;
                    
                }
                
            }
            consider(low);
            
        }
        ++cnt[last];
        
        
        
    }
    cout<<answer<<endl;
    
        
        
}    
   
return 0;
	

}
