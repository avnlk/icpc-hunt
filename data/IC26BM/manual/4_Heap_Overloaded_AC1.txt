#include <bits/stdc++.h>
using namespace std;
void s(){
    int c[4];
    cin>>c[0]>>c[1]>>c[2]>>c[3];
    string a="";
    for(int l=0;l<4;l++){
        if(c[l]==0) continue;
        int t[4];
        for(int i=0;i<4;i++) t[i]=c[i];
        t[l]--;
        int p1=min(t[0],t[3]);
        t[0]-=p1; t[3]-=p1;
        int p2=min(t[0],t[1]);
        t[0]-=p2; t[1]-=p2;
        int p3=min(t[1],t[3]);
        t[1]-=p3; t[3]-=p3;
        int p4=t[1]/2;
        t[1]-=p4*2;
        int z=2*(p1+p2)+t[0]+t[2];
        int n01=p3+p4;
        int o=t[1]+t[3];
        string r="";
        r.reserve(z+n01*2+o+2);
        r.append(z,'0');
        for(int i=0;i<n01;i++) r+="01";
        r.append(o,'1');
        if(l==0) r+="00";
        else if(l==1) r+="01";
        else if(l==2) r+="10";
        else if(l==3) r+="11";
        if(a=="" or r<a) a=r;
    }
    cout<<a<<"\n";
}
int main() {
	int t;
	cin>>t;
	while(t--) s();
	return 0;
};
