t=int(input())
for _ in range(t):
    c=list(map(int,input().split()))
    n=sum(c)
    tp=['00','01','10','11']
    ans=None
    for l in range(4):
        if c[l]==0:
            continue
        a=c.copy()
        a[l]-=1
        c00,c01,c10,c11=a
        extra=min(c00,c01+c11)
        z=c00+c10+extra
        ones01=min(c01,(c01+c11-extra)//2)
        ones10=min(c10,c00-extra,(c01+c11-extra-2*ones01)//2)
        ones=n-1-z-2*ones01-2*ones10
        cur=("0"*z+"01"*ones01+"10"*ones10+"1"*ones+tp[l])
        
        if ans is None or cur<ans:
            ans=cur
    print(ans)
