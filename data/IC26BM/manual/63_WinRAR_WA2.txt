#include <bits/stdc++.h>
using namespace std;

string ret(int a, int b, int c, int d) {
    string s = "";
    while (a > 0 && (d > 0 || b > 0)) {
        s += "00";
        --a;
        if (d >= b) --d;
        else --b;
    }
    while (a > 0) {
        s += "0";
        --a;
    }
    while (c > 0) {
        s += "0";
        --c;
    }
    while (b > 0 && d > 0) {
        s += "01";
        --b;
        --d;
    }
    while (b > 0) {
        s += "1";
        --b;
    }
    while (d > 0) {
        s += "1";
        --d;
    }
    return s;
}

int main() {
	// your code goes here
    int t = 0;
    cin >> t;
    while (t--) {
        int a, b, c, d;
        cin >> a >> b >> c >> d;
        string ans = "2";
        if (a > 0) {
            string s = ret(a-1, b, c, d) + "00";
            if (s < ans) ans = s;
        }
        if (b > 0) {
            string s = ret(a, b-1, c, d) + "01";
            if (s < ans) ans = s;
        }
        if (c > 0) {
            string s = ret(a, b, c-1, d) + "10";
            if (s < ans) ans = s;
        }
        if (d > 0) {
            string s = ret(a, b, c, d-1) + "11";
            if (s < ans) ans = s;
        }
        cout << ans << "\n";
    }
}
