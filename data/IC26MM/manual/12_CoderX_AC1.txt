#include <bits/stdc++.h>
using namespace std;

using vi= vector<int>;
#define pb push_back
#define all(x) (x).begin(),(x).end()
#define F(i,j,k) for(int i=(j);i<(k);i++)
#define EACH(x,a) for(auto &x:a)
#define endl '\n'

const int M=998244353;

void solve(){
    int n;cin>>n;
    vector<vi> p(n+1);
    F(i,1,n+1){
        int x;
        cin>>x;
        p[x].pb(i);
    }
    auto g= [&](int x,int l,int s){
        int r=x-1-l;
        int q=s?n-r:l+1;
        EACH(i,p[x]){
            int a=min(i,l)+max(0,i-n+r);
            int b=max(0,l-i+1)+min(r,n-i+1);
            
            if(!((q<=i && a==(i-1)/2)||(q>=i && b==(n-i)/2))){
                return 0;
            }
        }
        return 1;
    };
    vi d(n+1),e(n+1);
    d[0]=1;
    F(x,1,n+1){
        fill(all(e),0);
        F(l,0,x){
            if(!d[l]) continue;
            F(s,0,(x<n?2:1)){
                if(!g(x,l,s)) continue;
                
                int &z = e[l+!s];
                z+=d[l];
                
                if(z>=M) z-=M;   
                }
        }
        d.swap(e);
    }
    int a=0;
    EACH(x,d){
        a+=x;
        if(a>=M) a-=M;
    }
    
    cout<<a<<endl;
    
}


signed main() {
	// your code goes here
	ios_base::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	
	int t=1;
	cin>>t;
	while(t--){
	    solve();
	}

}
