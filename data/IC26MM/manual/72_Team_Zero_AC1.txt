#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define vec vector<ll>
#define MOD 998244353

bool valid( vector<vector<int>>& pos, int n, int value, int L, int R, int place )
{
    for ( int i : pos[value] )
    {
        int greater_prefix = 0;
        if ( L <= R && i >= L ) greater_prefix = max( 0, min( R, i ) - L + 1 );
        bool prefix_ok = ( place <= i && greater_prefix == i / 2 );
        
        int len = n - i + 1;
        int greater_sufix = 0;
        if ( L <= R && i <= R ) greater_sufix = max( 0, R - max( L, i ) + 1 );
        bool suffix_ok = ( place >= i && greater_sufix == len / 2 );
        if ( !prefix_ok && !suffix_ok ) return false;
    }
    return true;
}

int main() {
	int t;
	cin >> t;
	while ( t-- )
	{
	    int n;
	    cin >> n;
	    vector<int> a( n + 1 );
	    vector<vector<int>> pos( n + 1 );
	    for ( int i = 1; i <= n; i++ )
	    {
	        cin >> a[i];
	        pos[a[i]].push_back( i );
	    }
	    vector<ll> dp( n + 2, 0 );
	    for ( int i = 1; i <= n; i++ )
	    {
	        if ( valid( pos, n, n, 1, 0, i ) ) dp[i] = 1;
	    }
	    for ( int value = n - 1; value >= 1; value-- )
	    {
	        vector<ll> next( n + 2, 0 );
	        int len = n - value;
	        for ( int L = 1; L + len - 1 <= n; L++ )
	        {
	            if ( dp[L] == 0 ) continue;
	            
	            int R = L + len - 1;
	            
	            if ( L > 1 && valid( pos, n, value, L, R, L - 1 ) )
	            {
	                next[L - 1] += dp[L];
	                next[L - 1]  %= MOD;
	            }
	            if ( R < n && valid( pos, n, value, L, R, R + 1 ) )
	            {
	                next[L] += dp[L];
	                next[L] %= MOD;
	            }
	        }
	        dp = next;
	    }
	    ll ans = 0;
	    
	    for ( ll x : dp )
	    {
	        ans += x;
	        ans %= MOD;
	    }
	    cout << ans << '\n';
	}
    return 0;
}
