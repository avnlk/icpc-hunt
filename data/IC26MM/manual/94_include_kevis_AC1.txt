const int mod = 998244353;
int n;
vector<int>arr;
vector<vector<int>>dp;
vector<vector<int>>ps;

bool check(int x, int l, int r, int p){
    for(int i = 0; i < ps[x].size(); i++){
        int id = ps[x][i];
        bool k = false;
        if(p <= id && max(0, min(id, r) - l + 1) == (id/2) + 1) k = true;
        if(p >= id && max(0, r - max(id, l) + 1) == ((n - id + 1)/2) + 1) k = true;
        if(!k)return false;
        
    }
    return true;
}
int mem(int x, int l){
    if(x == n + 1) return 1;
    if(dp[x][l] != -1) return dp[x][l];
    int r = l + n - x;
    int ans = 0;
    if(check(x, l , r, l))
    {ans = (ans + mem(x+ 1, l+1))%mod;}
    if(x < n && check(x, l , r, r)) ans = (ans + mem(x + 1, l))%mod;
    return dp[x][l] =  ans;
    
}
