import sys

def solve():
    data=sys.stdin.read().split()
    if not data:
        return
    MOD=998244353
    MAXN=200005
    inv=[0]*MAXN
    inv[1]=1
    for i in range(2,MAXN):
        inv[i]=(MOD - MOD // i) * inv[MOD%i] %MOD
        
    h=[0]*(MAXN)
    for i in range(1,MAXN):
        h[i]=(h[i-1] + inv[i])% MOD
    
    it=iter(data)
    t=int(next(it))
    out=[]
    for _ in range(t):
        n=int(next(it))
        a=[int(next(it)) for _ in range(n)]
        total=0 
        for i in range(n):
            weight=(h[i]+ h[n-1-i]) %MOD
            total = (total+a[i]*weight) % MOD
        out.append(str(total))
    sys.stdout.write('\n'.join(out))
    
if __name__ == "__main__":
    solve()
