import sys
input=sys.stdin.readline

MOD=998244353
N=2000000

inv=[0]*(N+1)
h=[0]*(N+1)
inv[1]=1

for i in range(2,N+1):
    inv[i]=MOD-(MOD//i)*inv[MOD%i]%MOD
for i in range(1,N+1):
    h[i]=h[i-1]+inv[i]%MOD

for _ in range(int(input())):
    n=int(input())
    a=list(map(int,input().split()))
    ans=()
    
for i,x in enumerate(a):
    ans+=x*(tuple(h[i]+h[n-1-i]))
    
print(ans%MOD)
